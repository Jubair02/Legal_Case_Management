    76→    76→
    77→    77→export interface CaseFormDialogProps {
    78→    78→  open: boolean
    79→    79→  onOpenChange: (open: boolean) => void
    80→    80→  onSaved: () => void
    81→    81→  editing?: CaseDetailDTO | CaseListDTO | null
    82→    82→  /** Role of the acting user — LAWYER cannot reassign client/lawyer. */
    83→    83→  actorRole?: string
    84→    84→}
    85→    85→
    86→    86→export function CaseFormDialog({ open, onOpenChange, onSaved, editing, actorRole }: CaseFormDialogProps) {
    87→    87→  const [clients, setClients] = useState<ClientDTO[]>([])
    88→    88→  const [lawyers, setLawyers] = useState<LawyerDTO[]>([])
    89→    89→  const [optionsLoading, setOptionsLoading] = useState(false)
    90→    90→  const [pending, setPending] = useState(false)
    91→    91→
    92→    92→  const [caseNumber, setCaseNumber] = useState("")
    93→    93→  const [title, setTitle] = useState("")
    94→    94→  const [type, setType] = useState<string>("")
    95→    95→  const [clientId, setClientId] = useState<string>("")
    96→    96→  const [lawyerId, setLawyerId] = useState<string>("")
    97→    97→  const [court, setCourt] = useState<string>("")
    98→    98→  const [district, setDistrict] = useState<string>("")
    99→    99→  const [filingDate, setFilingDate] = useState("")
   100→   100→  const [status, setStatus] = useState<string>("ACTIVE")
   101→   101→  const [priority, setPriority] = useState<string>("MEDIUM")
   102→   102→  const [oppositeParty, setOppositeParty] = useState("")
   103→   103→  const [description, setDescription] = useState("")
   104→   104→  const [resolutionSummary, setResolutionSummary] = useState("")
   105→   105→  const [outcome, setOutcome] = useState("")
   106→   106→
   107→   107→  const isEditing = Boolean(editing)
   108→   108→  const isLawyer = actorRole === "LAWYER"
   109→   109→  const closing = status === "RESOLVED" || status === "CLOSED"
   110→   110→  /** Status options: restricted on create, full list when editing. */
   111→   111→  const statusOptions: readonly string[] = isEditing ? CASE_STATUSES : (["DRAFT", "ACTIVE", "PENDING"] as const)
   112→   112→
   113→   113→  // Prefill when the dialog opens
   114→   114→  useEffect(() => {
   115→   115→    if (!open) return
   116→   116→    const e = editing ?? null
   117→   117→    setCaseNumber(e?.caseNumber ?? "")
   118→   118→    setTitle(e?.title ?? "")
   119→   119→    setType(e?.type ?? "")
   120→   120→    setClientId(e?.client?.id ?? "")
   121→   121→    setLawyerId(e?.lawyer?.id ?? "")
   122→   122→    setCourt(e?.court ?? "")
   123→   123→    setDistrict(e?.district ?? "")
   124→   124→    setFilingDate(toDateInputValue(e?.filingDate))
   125→   125→    setStatus(e?.status ?? "ACTIVE")
   126→   126→    setPriority(e?.priority ?? "MEDIUM")
   127→   127→    setOppositeParty(e?.oppositeParty ?? "")
   128→   128→    setDescription(e && "description" in e ? (e.description ?? "") : "")
   129→   129→    setResolutionSummary(e && "resolutionSummary" in e ? (e.resolutionSummary ?? "") : "")
   130→   130→    setOutcome(e && "outcome" in e ? (e.outcome ?? "") : "")
   131→   131→  }, [open, editing])
   132→   132→
   133→   133→  // Load client / lawyer options when opened
   134→   134→  useEffect(() => {
   135→   135→    if (!open) return
   136→   136→    let active = true
   137→   137→    setOptionsLoading(true)
   138→   138→    Promise.all([apiGet<ClientDTO[]>("/api/clients"), apiGet<LawyerDTO[]>("/api/lawyers")])
   139→   139→      .then(([cls, lws]) => {
   140→   140→        if (!active) return
   141→   141→        setClients(Array.isArray(cls) ? cls : [])
   142→   142→        setLawyers(Array.isArray(lws) ? lws : [])
   143→   143→      })
   144→   144→      .catch((e: unknown) => {
   145→   145→        toast.error(e instanceof Error ? e.message : "Could not load clients and lawyers.")
   146→   146→      })
   147→   147→      .finally(() => {
   148→   148→        if (active) setOptionsLoading(false)
   149→   149→      })
   150→   150→    return () => {
   151→   151→      active = false
   152→   152→    }
   153→   153→  }, [open])
   154→   154→
   155→   155→  const handleSubmit = async () => {
   156→   156→    // Client-side validation
   157→   157→    if (!isEditing && !caseNumber.trim()) {
   158→   158→      toast.error("Case number is required.")
   159→   159→      return
   160→   160→    }
   161→   161→    if (!title.trim()) {
   162→   162→      toast.error("Case title is required.")
   163→   163→      return
   164→   164→    }
   165→   165→    if (!type) {
   166→   166→      toast.error("Please select a case type.")
   167→   167→      return
   168→   168→    }
   169→   169→    if (!clientId) {
   170→   170→      toast.error("Please select a client.")
   171→   171→      return
   172→   172→    }
   173→   173→    if (!court) {
   174→   174→      toast.error("Please select a court.")
   175→   175→      return
   176→   176→    }
   177→   177→    if (closing && !resolutionSummary.trim()) {
   178→   178→      toast.error("Resolution summary is required to resolve or close a case.")
   179→   179→      return
   180→   180→    }
   181→   181→
   182→   182→    const prev = editing ?? null
   183→   183→    const payload: Record<string, unknown> = {}
   184→   184→
   185→   185→    if (!prev) {
   186→   186→      // Create — required fields + provided optionals
   187→   187→      payload.caseNumber = caseNumber.trim()
   188→   188→      payload.title = title.trim()
   189→   189→      payload.type = type
   190→   190→      payload.clientId = clientId
   191→   191→      payload.court = court
   192→   192→      if (lawyerId) payload.lawyerId = lawyerId
   193→   193→      if (district) payload.district = district
   194→   194→      if (filingDate) payload.filingDate = filingDate
   195→   195→      payload.status = status
   196→   196→      payload.priority = priority
   197→   197→      if (oppositeParty.trim()) payload.oppositeParty = oppositeParty.trim()
   198→   198→      if (description.trim()) payload.description = description.trim()
   199→   199→      if (closing) payload.resolutionSummary = resolutionSummary.trim()
   200→   200→      if (outcome.trim()) payload.outcome = outcome.trim()
   201→   201→    } else {
   202→   202→      // Edit — only fields that were provided / actually changed.
   203→   203→      // District / filing date / opposite party / description are editable by
   204→   204→      // every role that can open this dialog (the server only blocks
   205→   205→      // assignment changes for lawyers), so they are always diffed and sent.
   206→   206→      payload.title = title.trim()
   207→   207→      payload.type = type
   208→   208→      payload.court = court
   209→   209→      payload.status = status
   210→   210→      payload.priority = priority
   211→   211→      if (!isLawyer) {
   212→   212→        // Only ADMIN/STAFF can change case assignment.
   213→   213→        if (clientId !== prev.client?.id) payload.clientId = clientId
   214→   214→        if ((lawyerId || null) !== (prev.lawyer?.id ?? null)) payload.lawyerId = lawyerId || null
   215→   215→      }
   216→   216→      if (district !== (prev.district ?? "")) payload.district = district.trim() || null
   217→   217→      if ((filingDate || null) !== (prev.filingDate ? toDateInputValue(prev.filingDate) : null))
   218→   218→        payload.filingDate = filingDate || null
   219→   219→      if (oppositeParty.trim() !== (prev.oppositeParty ?? ""))
   220→   220→        payload.oppositeParty = oppositeParty.trim() || null
   221→   221→      const prevDescription = "description" in prev ? (prev.description ?? "") : ""
   222→   222→      if (description.trim() !== prevDescription) payload.description = description.trim() || null
   223→   223→      if (closing) {
   224→   224→        payload.resolutionSummary = resolutionSummary.trim()
   225→   225→        if (outcome.trim()) payload.outcome = outcome.trim()
   226→   226→      }
   227→   227→    }
   228→   228→
   229→   229→    try {
   230→   230→      setPending(true)
   231→   231→      if (prev) {
   232→   232→        await apiSend("PATCH", `/api/cases/${prev.id}`, payload)
   233→   233→        toast.success("Case updated")
   234→   234→      } else {
   235→   235→        await apiSend("POST", "/api/cases", payload)
   236→   236→        toast.success("Case registered")
   237→   237→      }
   238→   238→      onSaved()
   239→   239→      onOpenChange(false)
   240→   240→    } catch (e) {
   241→   241→      toast.error(e instanceof Error ? e.message : "Could not save the case.")
   242→   242→    } finally {
   243→   243→      setPending(false)
   244→   244→    }
   245→   245→  }
   246→   246→
   247→   247→  return (
   248→   248→    <Dialog open={open} onOpenChange={onOpenChange}>
   249→   249→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   250→   250→        <DialogHeader>
   251→   251→          <DialogTitle>{isEditing ? "Edit Case" : "Register New Case"}</DialogTitle>
   252→   252→          <DialogDescription>
   253→   253→            {isEditing
   254→   254→              ? "Update the case file details below."
   255→   255→              : "Fill in the details to register a new case file."}
   256→   256→          </DialogDescription>
   257→   257→        </DialogHeader>
   258→   258→
   259→   259→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   260→   260→          <div className="space-y-2">
   261→   261→            <Label htmlFor="case-number">Case Number</Label>
   262→   262→            <Input
   263→   263→              id="case-number"
   264→   264→              value={caseNumber}
   265→   265→              onChange={(e) => setCaseNumber(e.target.value)}
   266→   266→              placeholder="e.g. CS-124/2026"
   267→   267→              disabled={isEditing}
   268→   268→              className="font-mono"
   269→   269→            />
   270→   270→          </div>
   271→   271→          <div className="space-y-2">
   272→   272→            <Label htmlFor="case-title">
   273→   273→              Case Title <span className="text-rose-600">*</span>
   274→   274→            </Label>
   275→   275→            <Input
   276→   276→              id="case-title"
   277→   277→              value={title}
   278→   278→              onChange={(e) => setTitle(e.target.value)}
   279→   279→              placeholder="e.g. Rahman vs Karim — land dispute"
   280→   280→            />
   281→   281→          </div>
   282→   282→
   283→   283→          <div className="space-y-2">
   284→   284→            <Label>
   285→   285→              Case Type <span className="text-rose-600">*</span>
   286→   286→            </Label>
   287→   287→            <Select value={type || undefined} onValueChange={setType}>
   288→   288→              <SelectTrigger className="w-full">
   289→   289→                <SelectValue placeholder="Select type" />
   290→   290→              </SelectTrigger>
   291→   291→              <SelectContent>
   292→   292→                {CASE_TYPES.map((t) => (
   293→   293→                  <SelectItem key={t} value={t}>
   294→   294→                    {t}
   295→   295→                  </SelectItem>
   296→   296→                ))}
   297→   297→              </SelectContent>
   298→   298→            </Select>
   299→   299→          </div>
   300→   300→
   301→   301→          <div className="space-y-2">
   302→   302→            <Label>
   303→   303→              Client <span className="text-rose-600">*</span>
   304→   304→            </Label>
   305→   305→            <Select
   306→   306→              value={clientId || undefined}
   307→   307→              onValueChange={setClientId}
   308→   308→              disabled={isLawyer}
   309→   309→            >
   310→   310→              <SelectTrigger className="w-full">
   311→   311→                <SelectValue placeholder={optionsLoading ? "Loading clients…" : "Select client"} />
   312→   312→              </SelectTrigger>
   313→   313→              <SelectContent>
   314→   314→                {clients.map((c) => (
   315→   315→                  <SelectItem key={c.id} value={c.id}>
   316→   316→                    {c.name}
   317→   317→                  </SelectItem>
   318→   318→                ))}
   319→   319→              </SelectContent>
   320→   320→            </Select>
   321→   321→          </div>
   322→   322→
   323→   323→          <div className="space-y-2">
   324→   324→            <Label>Assigned Lawyer</Label>
   325→   325→            <Select
   326→   326→              value={lawyerId || NONE}
   327→   327→              onValueChange={(v) => setLawyerId(v === NONE ? "" : v)}
   328→   328→              disabled={isLawyer}
   329→   329→            >
   330→   330→              <SelectTrigger className="w-full">
   331→   331→                <SelectValue placeholder={optionsLoading ? "Loading lawyers…" : "Unassigned"} />
   332→   332→              </SelectTrigger>
   333→   333→              <SelectContent>
   334→   334→                <SelectItem value={NONE}>Unassigned</SelectItem>
   335→   335→                {lawyers.map((l) => (
   336→   336→                  <SelectItem key={l.id} value={l.id}>
   337→   337→                    {l.name}
   338→   338→                  </SelectItem>
   339→   339→                ))}
   340→   340→              </SelectContent>
   341→   341→            </Select>
   342→   342→          </div>
   343→   343→
   344→   344→          <div className="space-y-2">
   345→   345→            <Label>
   346→   346→              Court <span className="text-rose-600">*</span>
   347→   347→            </Label>
   348→   348→            <Select value={court || undefined} onValueChange={setCourt}>
   349→   349→              <SelectTrigger className="w-full">
   350→   350→                <SelectValue placeholder="Select court" />
   351→   351→              </SelectTrigger>
   352→   352→              <SelectContent>
   353→   353→                {COURTS.map((ct) => (
   354→   354→                  <SelectItem key={ct} value={ct}>
   355→   355→                    {ct}
   356→   356→                  </SelectItem>
   357→   357→                ))}
   358→   358→              </SelectContent>
   359→   359→            </Select>
   360→   360→          </div>
   361→   361→
   362→   362→          <div className="space-y-2">
   363→   363→            <Label>District</Label>
   364→   364→            <Select value={district || NONE} onValueChange={(v) => setDistrict(v === NONE ? "" : v)}>
   365→   365→              <SelectTrigger className="w-full">
   366→   366→                <SelectValue placeholder="Select district" />
   367→   367→              </SelectTrigger>
   368→   368→              <SelectContent>
   369→   369→                <SelectItem value={NONE}>Not specified</SelectItem>
   370→   370→                {DISTRICTS.map((d) => (
   371→   371→                  <SelectItem key={d} value={d}>
   372→   372→                    {d}
   373→   373→                  </SelectItem>
   374→   374→                ))}
   375→   375→              </SelectContent>
   376→   376→            </Select>
   377→   377→          </div>
   378→   378→
   379→   379→          <div className="space-y-2">
   380→   380→            <Label htmlFor="filing-date">Filing Date</Label>
   381→   381→            <Input
   382→   382→              id="filing-date"
   383→   383→              type="date"
   384→   384→              value={filingDate}
   385→   385→              onChange={(e) => setFilingDate(e.target.value)}
   386→   386→            />
   387→   387→          </div>
   388→   388→
   389→   389→          <div className="space-y-2">
   390→   390→            <Label>Status</Label>
   391→   391→            <Select value={status} onValueChange={setStatus}>
   392→   392→              <SelectTrigger className="w-full">
   393→   393→                <SelectValue />
   394→   394→              </SelectTrigger>
   395→   395→              <SelectContent>
   396→   396→                {statusOptions.map((s) => (
   397→   397→                  <SelectItem key={s} value={s}>
   398→   398→                    {CASE_STATUS_LABELS[s] ?? s}
   399→   399→                  </SelectItem>
   400→   400→                ))}
   401→   401→              </SelectContent>
   402→   402→            </Select>
   403→   403→          </div>
   404→   404→
   405→   405→          <div className="space-y-2">
   406→   406→            <Label>Priority</Label>
   407→   407→            <Select value={priority} onValueChange={setPriority}>
   408→   408→              <SelectTrigger className="w-full">
   409→   409→                <SelectValue />
   410→   410→              </SelectTrigger>
   411→   411→              <SelectContent>
   412→   412→                {CASE_PRIORITIES.map((p) => (
   413→   413→                  <SelectItem key={p} value={p}>
   414→   414→                    {PRIORITY_LABELS[p] ?? p}
   415→   415→                  </SelectItem>
   416→   416→                ))}
   417→   417→              </SelectContent>
   418→   418→            </Select>
   419→   419→          </div>
   420→   420→
   421→   421→          <div className="space-y-2">
   422→   422→            <Label htmlFor="opposite-party">Opposite Party</Label>
   423→   423→            <Input
   424→   424→              id="opposite-party"
   425→   425→              value={oppositeParty}
   426→   426→              onChange={(e) => setOppositeParty(e.target.value)}
   427→   427→              placeholder="Name of the opposing party"
   428→   428→            />
   429→   429→          </div>
   430→   430→
   431→   431→          <div className="space-y-2 sm:col-span-2">
   432→   432→            <Label htmlFor="case-description">Description</Label>
   433→   433→            <Textarea
   434→   434→              id="case-description"
   435→   435→              value={description}
   436→   436→              onChange={(e) => setDescription(e.target.value)}
   437→   437→              placeholder="Brief summary of the matter…"
   438→   438→              rows={3}
   439→   439→            />
   440→   440→          </div>
   441→   441→
   442→   442→          {closing ? (
   443→   443→            <>
   444→   444→              <div className="space-y-2 sm:col-span-2">
   445→   445→                <Label htmlFor="resolution-summary">
   446→   446→                  Resolution Summary <span className="text-rose-600">*</span>
   447→   447→                </Label>
   448→   448→                <Textarea
   449→   449→                  id="resolution-summary"
   450→   450→                  value={resolutionSummary}
   451→   451→                  onChange={(e) => setResolutionSummary(e.target.value)}
   452→   452→                  placeholder="How was the case resolved?"
   453→   453→                  rows={3}
   454→   454→                  required
   455→   455→                />
   456→   456→                <p className="text-xs text-muted-foreground">
   457→   457→                  Example: Case resolved through mutual settlement.
   458→   458→                </p>
   459→   459→              </div>
   460→   460→              <div className="space-y-2">
   461→   461→                <Label htmlFor="case-outcome">Outcome</Label>
   462→   462→                <Input
   463→   463→                  id="case-outcome"
   464→   464→                  value={outcome}
   465→   465→                  onChange={(e) => setOutcome(e.target.value)}
   466→   466→                  placeholder="e.g. Won, Settled, Dismissed"
   467→   467→                />
   468→   468→              </div>
   469→   469→            </>
   470→   470→          ) : null}
   471→   471→        </div>
   472→   472→
   473→   473→        <DialogFooter>
   474→   474→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   475→   475→            Cancel
   476→   476→          </Button>
   477→   477→          <Button onClick={() => void handleSubmit()} disabled={pending}>
   478→   478→            {pending ? <Loader2 className="h-4 w-4 animate-spin" /> : null}
   479→   479→            {isEditing ? "Save Changes" : "Register Case"}
   480→   480→          </Button>
   481→   481→        </DialogFooter>
   482→   482→      </DialogContent>
   483→   483→    </Dialog>
   484→   484→  )
   485→   485→}
