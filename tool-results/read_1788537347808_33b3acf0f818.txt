   105→function RecordPaymentDialog({
   106→  invoice,
   107→  open,
   108→  onOpenChange,
   109→  onSaved,
   110→}: {
   111→  invoice: InvoiceDTO | null
   112→  open: boolean
   113→  onOpenChange: (open: boolean) => void
   114→  onSaved: () => void
   115→}) {
   116→  const [amount, setAmount] = useState("")
   117→  const [method, setMethod] = useState<string>(PAYMENT_METHODS[0])
   118→  const [paymentDate, setPaymentDate] = useState(toDateInputValue(new Date()))
   119→  const [reference, setReference] = useState("")
   120→  const [notes, setNotes] = useState("")
   121→  const [pending, setPending] = useState(false)
   122→
   123→  const remaining = invoice ? Math.max(0, num(invoice.amount) - num(invoice.paidAmount)) : 0
   124→
   125→  useEffect(() => {
   126→    if (open && invoice) {
   127→      setAmount(String(Math.max(0, num(invoice.amount) - num(invoice.paidAmount))))
   128→      setMethod(PAYMENT_METHODS[0])
   129→      setPaymentDate(toDateInputValue(new Date()))
   130→      setReference("")
   131→      setNotes("")
   132→    }
   133→  }, [open, invoice])
   134→
   135→  const submit = async () => {
   136→    if (!invoice) return
   137→    const amt = Number(amount)
   138→    if (!Number.isFinite(amt) || amt <= 0) {
   139→      toast.error("Enter an amount greater than 0.")
   140→      return
   141→    }
   142→    setPending(true)
   143→    try {
   144→      await apiSend("POST", "/api/payments", {
   145→        invoiceId: invoice.id,
   146→        amount: amt,
   147→        paymentMethod: method,
   148→        paymentDate: paymentDate || undefined,
   149→        referenceNumber: reference.trim() || undefined,
   150→        notes: notes.trim() || undefined,
   151→      })
   152→      toast.success("Payment recorded")
   153→      onSaved()
   154→      onOpenChange(false)
   155→    } catch (e) {
   156→      toast.error(errorMessage(e))
   157→    } finally {
   158→      setPending(false)
   159→    }
   160→  }
   161→
   162→  return (
   163→    <Dialog open={open} onOpenChange={onOpenChange}>
   164→      <DialogContent className="sm:max-w-lg">
   165→        <DialogHeader>
   166→          <DialogTitle>Record Payment</DialogTitle>
   167→          <DialogDescription>Record a client payment against this invoice.</DialogDescription>
   168→        </DialogHeader>
   169→
   170→        {invoice ? (
   171→          <div className="space-y-4">
   172→            <div className="space-y-1 rounded-lg border border-stone-200/80 bg-stone-50/60 p-3 text-sm">
   173→              <div className="flex items-center justify-between gap-2">
   174→                <span className="font-mono font-semibold">{invoice.invoiceNumber}</span>
   175→                <StatusBadge map={invoiceStatusStyles} value={invoice.status} />
   176→              </div>
   177→              <p className="text-muted-foreground">{invoice.clientName}</p>
   178→              <div className="flex flex-wrap gap-x-4 gap-y-1 pt-1 text-xs text-muted-foreground">
   179→                <span>
   180→                  Amount: <span className="font-semibold text-foreground">{formatCurrency(invoice.amount)}</span>
   181→                </span>
   182→                <span>
   183→                  Paid: <span className="font-semibold text-foreground">{formatCurrency(invoice.paidAmount)}</span>
   184→                </span>
   185→                <span>
   186→                  Remaining: <span className="font-bold text-emerald-700">{formatCurrency(remaining)}</span>
   187→                </span>
   188→              </div>
   189→            </div>
   190→
   191→            <div className="grid gap-4 sm:grid-cols-2">
   192→              <div className="space-y-2">
   193→                <Label htmlFor="payment-amount">Amount (৳) *</Label>
   194→                <Input
   195→                  id="payment-amount"
   196→                  type="number"
   197→                  min="0"
   198→                  step="0.01"
   199→                  inputMode="decimal"
   200→                  value={amount}
   201→                  onChange={(e) => setAmount(e.target.value)}
   202→                />
   203→              </div>
   204→              <div className="space-y-2">
   205→                <Label>Payment Method *</Label>
   206→                <Select value={method} onValueChange={setMethod}>
   207→                  <SelectTrigger className="w-full">
   208→                    <SelectValue placeholder="Select method" />
   209→                  </SelectTrigger>
   210→                  <SelectContent>
   211→                    {PAYMENT_METHODS.map((m) => (
   212→                      <SelectItem key={m} value={m}>
   213→                        {m}
   214→                      </SelectItem>
   215→                    ))}
   216→                  </SelectContent>
   217→                </Select>
   218→              </div>
   219→              <div className="space-y-2">
   220→                <Label htmlFor="payment-date">Payment Date</Label>
   221→                <Input
   222→                  id="payment-date"
   223→                  type="date"
   224→                  value={paymentDate}
   225→                  onChange={(e) => setPaymentDate(e.target.value)}
   226→                />
   227→              </div>
   228→              <div className="space-y-2">
   229→                <Label htmlFor="payment-reference">Reference Number</Label>
   230→                <Input
   231→                  id="payment-reference"
   232→                  value={reference}
   233→                  onChange={(e) => setReference(e.target.value)}
   234→                  placeholder="bKash TrxID / Bank ref"
   235→                />
   236→              </div>
   237→            </div>
   238→            <div className="space-y-2">
   239→              <Label htmlFor="payment-notes">Notes</Label>
   240→              <Textarea
   241→                id="payment-notes"
   242→                rows={2}
   243→                value={notes}
   244→                onChange={(e) => setNotes(e.target.value)}
   245→                placeholder="Optional note about this payment"
   246→              />
   247→            </div>
   248→          </div>
   249→        ) : null}
   250→
   251→        <DialogFooter>
   252→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   253→            Cancel
   254→          </Button>
   255→          <Button onClick={() => void submit()} disabled={pending}>
   256→            {pending ? "Saving…" : "Record Payment"}
   257→          </Button>
   258→        </DialogFooter>
   259→      </DialogContent>
   260→    </Dialog>
   261→  )
   262→}

   388→function NewInvoiceDialog({
   389→  open,
   390→  onOpenChange,
   391→  onSaved,
   392→}: {
   393→  open: boolean
   394→  onOpenChange: (open: boolean) => void
   395→  onSaved: () => void
   396→}) {
   397→  const clientsQ = useApiData<ClientDTO[]>(open ? "/api/clients" : null)
   398→  const casesQ = useApiData<CaseListDTO[]>(open ? "/api/cases" : null)
   399→
   400→  const [clientId, setClientId] = useState("")
   401→  const [caseId, setCaseId] = useState(NO_CASE)
   402→  const [billingType, setBillingType] = useState<string>(BILLING_TYPES[0])
   403→  const [description, setDescription] = useState("")
   404→  const [amount, setAmount] = useState("")
   405→  const [dueDate, setDueDate] = useState("")
   406→  const [pending, setPending] = useState(false)
   407→
   408→  useEffect(() => {
   409→    if (open) {
   410→      setClientId("")
   411→      setCaseId(NO_CASE)
   412→      setBillingType(BILLING_TYPES[0])
   413→      setDescription("")
   414→      setAmount("")
   415→      setDueDate("")
   416→    }
   417→  }, [open])
   418→
   419→  const onCaseChange = (value: string) => {
   420→    setCaseId(value)
   421→    if (value !== NO_CASE && casesQ.data) {
   422→      const kase = casesQ.data.find((c) => c.id === value)
   423→      if (kase?.client?.id) setClientId(kase.client.id)
   424→    }
   425→  }
   426→
   427→  const submit = async () => {
   428→    if (!clientId) {
   429→      toast.error("Please select a client.")
   430→      return
   431→    }
   432→    const amt = Number(amount)
   433→    if (!Number.isFinite(amt) || amt <= 0) {
   434→      toast.error("Amount must be greater than 0.")
   435→      return
   436→    }
   437→    setPending(true)
   438→    try {
   439→      await apiSend("POST", "/api/invoices", {
   440→        clientId,
   441→        caseId: caseId !== NO_CASE && caseId ? caseId : undefined,
   442→        billingType,
   443→        description: description.trim() || undefined,
   444→        amount: amt,
   445→        dueDate: dueDate || undefined,
   446→      })
   447→      toast.success("Invoice created")
   448→      onSaved()
   449→      onOpenChange(false)
   450→    } catch (e) {
   451→      toast.error(errorMessage(e))
   452→    } finally {
   453→      setPending(false)
   454→    }
   455→  }
   456→
   457→  return (
   458→    <Dialog open={open} onOpenChange={onOpenChange}>
   459→      <DialogContent className="sm:max-w-lg">
   460→        <DialogHeader>
   461→          <DialogTitle>New Invoice</DialogTitle>
   462→          <DialogDescription>Issue a new invoice to a client.</DialogDescription>
   463→        </DialogHeader>
   464→
   465→        <div className="space-y-4">
   466→          <div className="space-y-2">
   467→            <Label>Client *</Label>
   468→            <Select value={clientId || undefined} onValueChange={setClientId}>
   469→              <SelectTrigger className="w-full">
   470→                <SelectValue placeholder={clientsQ.loading ? "Loading clients…" : "Select client"} />
   471→              </SelectTrigger>
   472→              <SelectContent>
   473→                {(clientsQ.data ?? []).map((c) => (
   474→                  <SelectItem key={c.id} value={c.id}>
   475→                    {c.name}
   476→                  </SelectItem>
   477→                ))}
   478→              </SelectContent>
   479→            </Select>
   480→          </div>
   481→
   482→          <div className="space-y-2">
   483→            <Label>Case (optional)</Label>
   484→            <Select value={caseId} onValueChange={onCaseChange}>
   485→              <SelectTrigger className="w-full">

   506→                </SelectTrigger>
   507→                <SelectContent>
   508→                  {BILLING_TYPES.map((t) => (

   519→                id="invoice-amount"
   520→                type="number"
   521→                min="0"

   523→                inputMode="decimal"
   524→                value={amount}

   526→                placeholder="0"
   527→              />
   528→            </div>
   529→          </div>
   530→
   531→          <div className="space-y-2">
   532→            <Label htmlFor="invoice-due">Due Date</Label>

   534→          </div>
   535→
   536→          <div className="space-y-2">
   537→            <Label htmlFor="invoice-description">Description</Label>
   538→            <Textarea
   539→              id="invoice-description"
   540→              rows={2}
   541→              value={description}
   542→              onChange={(e) => setDescription(e.target.value)}

   544→            />
   545→          </div>
   546→        </div>
   547→
   548→        <DialogFooter>
   549→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>

   551→          </Button>

   553→            {pending ? "Creating…" : "Create Invoice"}
   554→          </Button>
   555→        </DialogFooter>
   556→      </DialogContent>
   557→    </Dialog>

   563→function InvoicesTab({
   564→  invoices,
   565→  loading,
   566→  error,
   567→  isAdmin,
   568→  isClient,
   569→  navigate,
   570→  onRecordPayment,
   571→  onViewDetails,
   572→  onCancel,
   573→  onDelete,
   574→  onRetry,
   575→}: {
   576→  invoices: InvoiceDTO[] | null
   577→  loading: boolean
   578→  error: string | null
   579→  isAdmin: boolean
   580→  isClient: boolean
   581→  navigate: NavigateFn
   582→  onRecordPayment: (inv: InvoiceDTO) => void
   583→  onViewDetails: (inv: InvoiceDTO) => void
   584→  onCancel: (inv: InvoiceDTO) => void
   585→  onDelete: (inv: InvoiceDTO) => void
   586→  onRetry: () => void
   587→}) {
   588→  const [status, setStatus] = useState(ALL)
   589→  const [search, setSearch] = useState("")
   590→
   591→  const list = invoices ?? []
   592→  const active = useMemo(() => list.filter((i) => i.status !== "CANCELLED"), [list])

   603→        (i) => i.invoiceNumber.toLowerCase().includes(q) || i.clientName.toLowerCase().includes(q)

   619→            Try again

   621→        }
   622→      />

   641→            icon={Banknote}

   649→            label="Outstanding"
   650→            value={formatCurrency(Math.max(0, outstanding))}

   653→          />
   654→        </div>

   661→          <Input
   662→            value={search}
   663→            onChange={(e) => setSearch(e.target.value)}

   665→            className="pl-8"
   666→          />
   667→        </div>

   669→          <SelectTrigger className="w-full sm:w-44">
   670→            <SelectValue placeholder="All statuses" />
   671→          </SelectTrigger>

   673→            <SelectItem value={ALL}>All statuses</SelectItem>

   685→        {filtered.length === 0 ? (
   686→          <div className="p-6">
   687→            <EmptyState
   688→              icon={Receipt}
   689→              title={list.length === 0 ? "No invoices yet" : "No invoices match your filters"}
   690→              description={
   691→                list.length === 0
   692→                  ? "Invoices you create will appear here."
   693→                  : "Try a different search term or status filter."
   694→              }
   695→            />
   696→          </div>
   697→        ) : (
   698→          <Table>
   699→            <TableHeader>
   700→              <TableRow className="hover:bg-transparent">
   701→                <TableHead>Invoice</TableHead>
   702→                <TableHead>Case</TableHead>
   703→                <TableHead>Client</TableHead>
   704→                <TableHead>Type</TableHead>
   705→                <TableHead className="text-right">Amount</TableHead>
   706→                <TableHead className="text-right">Paid</TableHead>

   778→      </div>
