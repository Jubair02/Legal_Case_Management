    77→    77→export interface CaseFormDialogProps {

    85→    85→
    86→    86→export function CaseFormDialog({ open, onOpenChange, onSaved, editing, actorRole }: CaseFormDialogProps) {
    87→    87→  const [clients, setClients] = useState<ClientDTO[]>([])
    88→    88→  const [lawyers, setLawyers] = useState<LawyerDTO[]>([])
    89→    89→  const [optionsLoading, setOptionsLoading] = useState(false)
    90→    90→  const [pending, setPending] = useState(false)
    91→    91→
    92→    92→  const [caseNumber, setCaseNumber] = useState("")
    93→    93→  const [title, setTitle] = useState("")
    94→    94→  const [type, setType] = useState<string>("")
    95→    95→  const [clientId, setClientId] = useState<string>("")
    96→    96→  const [lawyerId, setLawyerId] = useState<string>("")
    97→    97→  const [court, setCourt] = useState<string>("")
    98→    98→  const [district, setDistrict] = useState<string>("")
    99→    99→  const [filingDate, setFilingDate] = useState("")
   100→   100→  const [status, setStatus] = useState<string>("ACTIVE")
   101→   101→  const [priority, setPriority] = useState<string>("MEDIUM")
   102→   102→  const [oppositeParty, setOppositeParty] = useState("")
   103→   103→  const [description, setDescription] = useState("")
   104→   104→  const [resolutionSummary, setResolutionSummary] = useState("")
   105→   105→  const [outcome, setOutcome] = useState("")
   106→   106→
   107→   107→  const isEditing = Boolean(editing)
   108→   108→  const isLawyer = actorRole === "LAWYER"
   109→   109→  const closing = status === "RESOLVED" || status === "CLOSED"
   110→   110→  /** Status options: restricted on create, full list when editing. */
   111→   111→  const statusOptions: readonly string[] = isEditing ? CASE_STATUSES : (["DRAFT", "ACTIVE", "PENDING"] as const)
   112→   112→
   113→   113→  // Prefill when the dialog opens
   114→   114→  useEffect(() => {
   115→   115→    if (!open) return
   116→   116→    const e = editing ?? null
   117→   117→    setCaseNumber(e?.caseNumber ?? "")
   118→   118→    setTitle(e?.title ?? "")
   119→   119→    setType(e?.type ?? "")
   120→   120→    setClientId(e?.client?.id ?? "")
   121→   121→    setLawyerId(e?.lawyer?.id ?? "")
   122→   122→    setCourt(e?.court ?? "")
   123→   123→    setDistrict(e?.district ?? "")
   124→   124→    setFilingDate(toDateInputValue(e?.filingDate))
   125→   125→    setStatus(e?.status ?? "ACTIVE")
   126→   126→    setPriority(e?.priority ?? "MEDIUM")

   128→   128→    setDescription(e && "description" in e ? (e.description ?? "") : "")

   132→   132→

   140→   140→        if (!active) return
   141→   141→        setClients(Array.isArray(cls) ? cls : [])
   142→   142→        setLawyers(Array.isArray(lws) ? lws : [])

   144→   144→      .catch((e: unknown) => {
   145→   145→        toast.error(e instanceof Error ? e.message : "Could not load clients and lawyers.")
   146→   146→      })
   147→   147→      .finally(() => {
   148→   148→        if (active) setOptionsLoading(false)
   149→   149→      })
   150→   150→    return () => {
   151→   151→      active = false
   152→   152→    }
   153→   153→  }, [open])
   154→   154→
   155→   155→  const handleSubmit = async () => {
   156→   156→    // Client-side validation
   157→   157→    if (!isEditing && !caseNumber.trim()) {
   158→   158→      toast.error("Case number is required.")
   159→   159→      return
   160→   160→    }
   161→   161→    if (!title.trim()) {
   162→   162→      toast.error("Case title is required.")
   163→   163→      return
   164→   164→    }
   165→   165→    if (!type) {
   166→   166→      toast.error("Please select a case type.")
   167→   167→      return
   168→   168→    }
   169→   169→    if (!clientId) {
   170→   170→      toast.error("Please select a client.")
   171→   171→      return
   172→   172→    }
   173→   173→    if (!court) {
   174→   174→      toast.error("Please select a court.")
   175→   175→      return
   176→   176→    }
   177→   177→    if (closing && !resolutionSummary.trim()) {
   178→   178→      toast.error("Resolution summary is required to resolve or close a case.")
   179→   179→      return
   180→   180→    }
   181→   181→
   182→   182→    const prev = editing ?? null
   183→   183→    const payload: Record<string, unknown> = {}
   184→   184→
   185→   185→    if (!prev) {
   186→   186→      // Create — required fields + provided optionals
   187→   187→      payload.caseNumber = caseNumber.trim()
   188→   188→      payload.title = title.trim()
   189→   189→      payload.type = type
   190→   190→      payload.clientId = clientId
   191→   191→      payload.court = court
   192→   192→      if (lawyerId) payload.lawyerId = lawyerId
   193→   193→      if (district) payload.district = district
   194→   194→      if (filingDate) payload.filingDate = filingDate
   195→   195→      payload.status = status
   196→   196→      payload.priority = priority
   197→   197→      if (oppositeParty.trim()) payload.oppositeParty = oppositeParty.trim()
   198→   198→      if (description.trim()) payload.description = description.trim()
   199→   199→      if (closing) payload.resolutionSummary = resolutionSummary.trim()
   200→   200→      if (outcome.trim()) payload.outcome = outcome.trim()
   201→   201→    } else {
   202→   202→      // Edit — only fields that were provided / actually changed
   203→   203→      payload.title = title.trim()
   204→   204→      payload.type = type
   205→   205→      payload.court = court
   206→   206→      payload.status = status
   207→   207→      payload.priority = priority
   208→   208→      if (!isLawyer) {
   209→   209→        if (clientId !== prev.client?.id) payload.clientId = clientId
   210→   210→        if ((lawyerId || null) !== (prev.lawyer?.id ?? null)) payload.lawyerId = lawyerId || null
   211→   211→        if (district !== (prev.district ?? "")) payload.district = district.trim() || null
   212→   212→        if ((filingDate || null) !== (prev.filingDate ? toDateInputValue(prev.filingDate) : null))
   213→   213→          payload.filingDate = filingDate || null
   214→   214→        if (oppositeParty.trim() !== (prev.oppositeParty ?? "")) payload.oppositeParty = oppositeParty.trim() || null
   215→   215→        const prevDescription = "description" in prev ? (prev.description ?? "") : ""
   216→   216→        if (description.trim() !== prevDescription) payload.description = description.trim() || null
   217→   217→      }
   218→   218→      if (closing) {
   219→   219→        payload.resolutionSummary = resolutionSummary.trim()
   220→   220→        if (outcome.trim()) payload.outcome = outcome.trim()
   221→   221→      }
   222→   222→    }
   223→   223→
   224→   224→    try {
   225→   225→      setPending(true)
   226→   226→      if (prev) {
   227→   227→        await apiSend("PATCH", `/api/cases/${prev.id}`, payload)
   228→   228→        toast.success("Case updated")
   229→   229→      } else {
   230→   230→        await apiSend("POST", "/api/cases", payload)
   231→   231→        toast.success("Case registered")
   232→   232→      }
   233→   233→      onSaved()
   234→   234→      onOpenChange(false)
   235→   235→    } catch (e) {
   236→   236→      toast.error(e instanceof Error ? e.message : "Could not save the case.")
   237→   237→    } finally {
   238→   238→      setPending(false)
   239→   239→    }
   240→   240→  }
   241→   241→
   242→   242→  return (
   243→   243→    <Dialog open={open} onOpenChange={onOpenChange}>
   244→   244→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   245→   245→        <DialogHeader>
   246→   246→          <DialogTitle>{isEditing ? "Edit Case" : "Register New Case"}</DialogTitle>
   247→   247→          <DialogDescription>
   248→   248→            {isEditing
   249→   249→              ? "Update the case file details below."
   250→   250→              : "Fill in the details to register a new case file."}
   251→   251→          </DialogDescription>
   252→   252→        </DialogHeader>
   253→   253→
   254→   254→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   255→   255→          <div className="space-y-2">
   256→   256→            <Label htmlFor="case-number">Case Number</Label>
   257→   257→            <Input
   258→   258→              id="case-number"
   259→   259→              value={caseNumber}
   260→   260→              onChange={(e) => setCaseNumber(e.target.value)}
   261→   261→              placeholder="e.g. CS-124/2026"
   262→   262→              disabled={isEditing}
   263→   263→              className="font-mono"
   264→   264→            />
   265→   265→          </div>
   266→   266→          <div className="space-y-2">
   267→   267→            <Label htmlFor="case-title">
   268→   268→              Case Title <span className="text-rose-600">*</span>
   269→   269→            </Label>
   270→   270→            <Input
   271→   271→              id="case-title"
   272→   272→              value={title}
   273→   273→              onChange={(e) => setTitle(e.target.value)}
   274→   274→              placeholder="e.g. Rahman vs Karim — land dispute"
   275→   275→            />
   276→   276→          </div>
   277→   277→
   278→   278→          <div className="space-y-2">
   279→   279→            <Label>
   280→   280→              Case Type <span className="text-rose-600">*</span>
   281→   281→            </Label>
   282→   282→            <Select value={type || undefined} onValueChange={setType}>
   283→   283→              <SelectTrigger className="w-full">
   284→   284→                <SelectValue placeholder="Select type" />
   285→   285→              </SelectTrigger>
   286→   286→              <SelectContent>
   287→   287→                {CASE_TYPES.map((t) => (
   288→   288→                  <SelectItem key={t} value={t}>
   289→   289→                    {t}
   290→   290→                  </SelectItem>
   291→   291→                ))}
   292→   292→              </SelectContent>
   293→   293→            </Select>
   294→   294→          </div>
   295→   295→
   296→   296→          <div className="space-y-2">
   297→   297→            <Label>
   298→   298→              Client <span className="text-rose-600">*</span>
   299→   299→            </Label>
   300→   300→            <Select
   301→   301→              value={clientId || undefined}

   303→   303→              disabled={isLawyer}
   304→   304→            >
   305→   305→              <SelectTrigger className="w-full">
   306→   306→                <SelectValue placeholder={optionsLoading ? "Loading clients…" : "Select client"} />
   307→   307→              </SelectTrigger>
   308→   308→              <SelectContent>
   309→   309→                {clients.map((c) => (
   310→   310→                  <SelectItem key={c.id} value={c.id}>
   311→   311→                    {c.name}
   312→   312→                  </SelectItem>
   313→   313→                ))}
   314→   314→              </SelectContent>
   315→   315→            </Select>
   316→   316→          </div>
   317→   317→
   318→   318→          <div className="space-y-2">
   319→   319→            <Label>Assigned Lawyer</Label>
   320→   320→            <Select
   321→   321→              value={lawyerId || NONE}
   322→   322→              onValueChange={(v) => setLawyerId(v === NONE ? "" : v)}
   323→   323→              disabled={isLawyer}
   324→   324→            >
   325→   325→              <SelectTrigger className="w-full">
   326→   326→                <SelectValue placeholder={optionsLoading ? "Loading lawyers…" : "Unassigned"} />
   327→   327→              </SelectTrigger>
   328→   328→              <SelectContent>
   329→   329→                <SelectItem value={NONE}>Unassigned</SelectItem>

   331→   331→                  <SelectItem key={l.id} value={l.id}>

   333→   333→                  </SelectItem>

   335→   335→              </SelectContent>
   336→   336→            </Select>
   337→   337→          </div>
   338→   338→
   339→   339→          <div className="space-y-2">

   363→   363→              <SelectContent>

   412→   412→              </SelectContent>

   414→   414→          </div>

   416→   416→          <div className="space-y-2">

   452→   452→                  Example: Case resolved through mutual settlement.

   457→   457→                <Input
   458→   458→                  id="case-outcome"
   459→   459→                  value={outcome}
   460→   460→                  onChange={(e) => setOutcome(e.target.value)}
   461→   461→                  placeholder="e.g. Won, Settled, Dismissed"
   462→   462→                />
   463→   463→              </div>

   466→   466→        </div>

   468→   468→        <DialogFooter>

   475→   475→          </Button>
   476→   476→        </DialogFooter>
   477→   477→      </DialogContent>
   478→   478→    </Dialog>
   479→   479→  )
   480→   480→}
