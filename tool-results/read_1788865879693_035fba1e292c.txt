   265→export interface LawyerFormDialogProps {
   266→  open: boolean
   267→  onOpenChange: (open: boolean) => void
   268→  /** When set the dialog edits this lawyer; otherwise it creates a new one. */
   269→  lawyer?: LawyerDTO | null
   270→  onSaved: () => void
   271→}
   272→
   273→export function LawyerFormDialog({ open, onOpenChange, lawyer, onSaved }: LawyerFormDialogProps) {
   274→  const isEdit = Boolean(lawyer)
   275→  const [name, setName] = useState("")
   276→  const [phone, setPhone] = useState("")
   277→  const [email, setEmail] = useState("")
   278→  const [barCouncilId, setBarCouncilId] = useState("")
   279→  const [specialization, setSpecialization] = useState("__none__")
   280→  const [chamberName, setChamberName] = useState("")
   281→  const [experience, setExperience] = useState("")
   282→  const [status, setStatus] = useState("ACTIVE")
   283→  const [portal, setPortal] = useState(false)
   284→  const [password, setPassword] = useState("")
   285→  const [pending, setPending] = useState(false)
   286→
   287→  useEffect(() => {
   288→    if (!open) return
   289→    setName(lawyer?.name ?? "")
   290→    setPhone(lawyer?.phone ?? "")
   291→    setEmail(lawyer?.email ?? "")
   292→    setBarCouncilId(lawyer?.barCouncilId ?? "")
   293→    setSpecialization(lawyer?.specialization && SPECIALIZATIONS.includes(lawyer.specialization as (typeof SPECIALIZATIONS)[number]) ? lawyer.specialization : "__none__")
   294→    setChamberName(lawyer?.chamberName ?? "")
   295→    setExperience(lawyer?.experience != null ? String(lawyer.experience) : "")
   296→    setStatus(lawyer?.status === "INACTIVE" ? "INACTIVE" : "ACTIVE")
   297→    setPortal(false)
   298→    setPassword("")
   299→    setPending(false)
   300→  }, [open, lawyer])
   301→
   302→  const submit = async () => {
   303→    if (!name.trim()) {
   304→      toast.error("Lawyer name is required.")
   305→      return
   306→    }
   307→    const expNum = experience.trim() === "" ? null : Number(experience)
   308→    if (expNum !== null && (Number.isNaN(expNum) || expNum < 0)) {
   309→      toast.error("Experience must be a non-negative number.")
   310→      return
   311→    }
   312→    if (!isEdit && portal) {
   313→      if (!email.trim() || !isValidEmail(email)) {
   314→        toast.error("Please enter a valid email address.")
   315→        return
   316→      }
   317→      if (password.length < 6) {
   318→        toast.error("Portal password must be at least 6 characters.")
   319→        return
   320→      }
   321→    }
   322→    const base = {
   323→      name: name.trim(),
   324→      phone: phone.trim() || null,
   325→      email: email.trim() || null,
   326→      barCouncilId: barCouncilId.trim() || null,
   327→      specialization: specialization === "__none__" ? null : specialization,
   328→      chamberName: chamberName.trim() || null,
   329→      experience: expNum,
   330→    }
   331→    try {
   332→      setPending(true)
   333→      if (isEdit && lawyer) {
   334→        await apiSend<LawyerDTO>("PATCH", `/api/lawyers/${lawyer.id}`, { ...base, status })
   335→        toast.success("Lawyer updated.")
   336→      } else {
   337→        await apiSend<LawyerDTO>("POST", "/api/lawyers", {
   338→          ...base,
   339→          ...(portal ? { createPortalAccess: true, password } : {}),
   340→        })
   341→        toast.success(portal ? "Lawyer created with portal access." : "Lawyer created.")
   342→      }
   343→      onOpenChange(false)
   344→      onSaved()
   345→    } catch (e) {
   346→      toast.error(e instanceof Error ? e.message : "Could not save the lawyer.")
   347→    } finally {
   348→      setPending(false)
   349→    }
   350→  }
   351→
   352→  return (
   353→    <Dialog open={open} onOpenChange={onOpenChange}>
   354→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   355→        <DialogHeader>
   356→          <DialogTitle>{isEdit ? "Edit lawyer" : "Add lawyer"}</DialogTitle>
   357→          <DialogDescription>
   358→            {isEdit ? "Update the advocate's credentials and status." : "Register an advocate of the chamber."}
   359→          </DialogDescription>
   360→        </DialogHeader>
   361→
   362→        <div className="space-y-4">
   363→          <div className="space-y-1.5">
   364→            <Label htmlFor="lawyer-name">
   365→              Name <span className="text-rose-500">*</span>
   366→            </Label>
   367→            <Input
   368→              id="lawyer-name"
   369→              value={name}
   370→              onChange={(e) => setName(e.target.value)}
   371→              placeholder="e.g. Kamal Hossain"
   372→              autoFocus
   373→            />
   374→          </div>
   375→
   376→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   377→            <div className="space-y-1.5">
   378→              <Label htmlFor="lawyer-phone">Phone</Label>
   379→              <Input id="lawyer-phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+8801XXXXXXXXX" />
   380→            </div>
   381→            <div className="space-y-1.5">
   382→              <Label htmlFor="lawyer-email">Email</Label>
   383→              <Input id="lawyer-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="lawyer@chamber.bd" />
   384→            </div>
   385→            <div className="space-y-1.5">
   386→              <Label htmlFor="lawyer-bar">Bar Council ID</Label>
   387→              <Input id="lawyer-bar" value={barCouncilId} onChange={(e) => setBarCouncilId(e.target.value)} placeholder="e.g. D-12345" />
   388→            </div>
   389→            <div className="space-y-1.5">
   390→              <Label htmlFor="lawyer-specialization">Specialization</Label>
   391→              <Select value={specialization} onValueChange={setSpecialization}>
   392→                <SelectTrigger id="lawyer-specialization" className="w-full">
   393→                  <SelectValue placeholder="Not specified" />
   394→                </SelectTrigger>
   395→                <SelectContent>
   396→                  <SelectItem value="__none__">Not specified</SelectItem>
   397→                  {SPECIALIZATIONS.map((s) => (
   398→                    <SelectItem key={s} value={s}>
   399→                      {s}
   400→                    </SelectItem>
   401→                  ))}
   402→                </SelectContent>
   403→              </Select>
   404→            </div>
   405→            <div className="space-y-1.5">
   406→              <Label htmlFor="lawyer-chamber">Chamber name</Label>
   407→              <Input id="lawyer-chamber" value={chamberName} onChange={(e) => setChamberName(e.target.value)} placeholder="e.g. Hossain & Associates" />
   408→            </div>
   409→            <div className="space-y-1.5">
   410→              <Label htmlFor="lawyer-experience">Experience (years)</Label>
   411→              <Input
   412→                id="lawyer-experience"
   413→                type="number"
   414→                min={0}
   415→                value={experience}
   416→                onChange={(e) => setExperience(e.target.value)}
   417→                placeholder="e.g. 12"
   418→              />
   419→            </div>
   420→            {isEdit ? (
   421→              <div className="space-y-1.5">
   422→                <Label htmlFor="lawyer-status">Status</Label>
   423→                <Select value={status} onValueChange={setStatus}>
   424→                  <SelectTrigger id="lawyer-status" className="w-full">
   425→                    <SelectValue placeholder="Select status" />
   426→                  </SelectTrigger>
   427→                  <SelectContent>
   428→                    <SelectItem value="ACTIVE">Active</SelectItem>
   429→                    <SelectItem value="INACTIVE">Inactive</SelectItem>
   430→                  </SelectContent>
   431→                </Select>
   432→              </div>
   433→            ) : null}
   434→          </div>
   435→
   436→          {!isEdit ? (
   437→            <div className="space-y-3 rounded-lg border border-stone-200 bg-stone-50 p-3">
   438→              <div className="flex items-center justify-between gap-3">
   439→                <div className="min-w-0">
   440→                  <Label htmlFor="lawyer-portal">Create lawyer portal account</Label>
   441→                  <p className="text-xs text-muted-foreground">Lawyer signs in with their email + this password</p>
   442→                </div>
   443→                <Switch id="lawyer-portal" checked={portal} onCheckedChange={setPortal} />
   444→              </div>
   445→              {portal ? (
   446→                <div className="space-y-1.5">
   447→                  <Label htmlFor="lawyer-portal-password">Portal password</Label>
   448→                  <Input
   449→                    id="lawyer-portal-password"
   450→                    type="password"
   451→                    value={password}
   452→                    onChange={(e) => setPassword(e.target.value)}
   453→                    placeholder="Minimum 6 characters"
   454→                    autoComplete="new-password"
   455→                  />
   456→                  <p className="text-xs text-muted-foreground">Minimum 6 characters.</p>
   457→                </div>
   458→              ) : null}
   459→            </div>
   460→          ) : null}
   461→        </div>
   462→
   463→        <DialogFooter>
   464→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   465→            Cancel
   466→          </Button>
   467→          <Button onClick={submit} disabled={pending}>
   468→            {pending ? "Saving…" : isEdit ? "Save changes" : "Create lawyer"}
   469→          </Button>
   470→        </DialogFooter>
   471→      </DialogContent>
   472→    </Dialog>
   473→  )
   474→}
