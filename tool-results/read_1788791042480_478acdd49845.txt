   104→
   105→function RecordPaymentDialog({
   106→  invoice,
   107→  open,
   108→  onOpenChange,
   109→  onSaved,
   110→}: {
   111→  invoice: InvoiceDTO | null
   112→  open: boolean
   113→  onOpenChange: (open: boolean) => void
   114→  onSaved: () => void
   115→}) {
   116→  const [amount, setAmount] = useState("")
   117→  const [method, setMethod] = useState<string>(PAYMENT_METHODS[0])
   118→  const [paymentDate, setPaymentDate] = useState(toDateInputValue(new Date()))
   119→  const [reference, setReference] = useState("")
   120→  const [notes, setNotes] = useState("")
   121→  const [pending, setPending] = useState(false)
   122→
   123→  const remaining = invoice ? Math.max(0, num(invoice.amount) - num(invoice.paidAmount)) : 0
   124→
   125→  useEffect(() => {
   126→    if (open && invoice) {
   127→      setAmount(String(Math.max(0, num(invoice.amount) - num(invoice.paidAmount))))
   128→      setMethod(PAYMENT_METHODS[0])
   129→      setPaymentDate(toDateInputValue(new Date()))
   130→      setReference("")
   131→      setNotes("")
   132→    }
   133→  }, [open, invoice])
   134→
   135→  const submit = async () => {
   136→    if (!invoice) return
   137→    const amt = Number(amount)
   138→    if (!Number.isFinite(amt) || amt <= 0) {
   139→      toast.error("Enter an amount greater than 0.")
   140→      return
   141→    }
   142→    if (amt > remaining + 0.005) {
   143→      toast.error(`Payment exceeds the remaining due of ${formatCurrency(remaining)}.`)
   144→      return
   145→    }
   146→    setPending(true)
   147→    try {
   148→      await apiSend("POST", "/api/payments", {
   149→        invoiceId: invoice.id,
   150→        amount: amt,
   151→        paymentMethod: method,
   152→        paymentDate: paymentDate || undefined,
   153→        referenceNumber: reference.trim() || undefined,
   154→        notes: notes.trim() || undefined,
   155→      })
   156→      toast.success("Payment recorded")
   157→      onSaved()
   158→      onOpenChange(false)
   159→    } catch (e) {
   160→      toast.error(errorMessage(e))
   161→    } finally {
   162→      setPending(false)
   163→    }
   164→  }
   165→
   166→  return (
   167→    <Dialog open={open} onOpenChange={onOpenChange}>
   168→      <DialogContent className="sm:max-w-lg">
   169→        <DialogHeader>
   170→          <DialogTitle>Record Payment</DialogTitle>
   171→          <DialogDescription>Record a client payment against this invoice.</DialogDescription>
   172→        </DialogHeader>
   173→
   174→        {invoice ? (
   175→          <div className="space-y-4">
   176→            <div className="space-y-1 rounded-lg border border-stone-200/80 bg-stone-50/60 p-3 text-sm">
   177→              <div className="flex items-center justify-between gap-2">
   178→                <span className="font-mono font-semibold">{invoice.invoiceNumber}</span>
   179→                <StatusBadge map={invoiceStatusStyles} value={invoice.status} />
   180→              </div>
   181→              <p className="text-muted-foreground">{invoice.clientName}</p>
   182→              <div className="flex flex-wrap gap-x-4 gap-y-1 pt-1 text-xs text-muted-foreground">
   183→                <span>
   184→                  Amount: <span className="font-semibold text-foreground">{formatCurrency(invoice.amount)}</span>
   185→                </span>
   186→                <span>
   187→                  Paid: <span className="font-semibold text-foreground">{formatCurrency(invoice.paidAmount)}</span>
   188→                </span>
   189→                <span>
   190→                  Remaining: <span className="font-bold text-emerald-700">{formatCurrency(remaining)}</span>
   191→                </span>
   192→              </div>
   193→            </div>
   194→
   195→            <div className="grid gap-4 sm:grid-cols-2">
   196→              <div className="space-y-2">
   197→                <Label htmlFor="payment-amount">Amount (৳) *</Label>
   198→                <Input
   199→                  id="payment-amount"
   200→                  type="number"
   201→                  min="0"
   202→                  step="0.01"
   203→                  inputMode="decimal"
   204→                  value={amount}
   205→                  onChange={(e) => setAmount(e.target.value)}
   206→                />
   207→              </div>
   208→              <div className="space-y-2">
   209→                <Label>Payment Method *</Label>
   210→                <Select value={method} onValueChange={setMethod}>
   211→                  <SelectTrigger className="w-full">
   212→                    <SelectValue placeholder="Select method" />
   213→                  </SelectTrigger>
   214→                  <SelectContent>
   215→                    {PAYMENT_METHODS.map((m) => (
   216→                      <SelectItem key={m} value={m}>
   217→                        {m}
   218→                      </SelectItem>
   219→                    ))}
   220→                  </SelectContent>
   221→                </Select>
   222→              </div>
   223→              <div className="space-y-2">
   224→                <Label htmlFor="payment-date">Payment Date</Label>
   225→                <Input
   226→                  id="payment-date"
   227→                  type="date"
   228→                  value={paymentDate}
   229→                  onChange={(e) => setPaymentDate(e.target.value)}
   230→                />
   231→              </div>
   232→              <div className="space-y-2">
   233→                <Label htmlFor="payment-reference">Reference Number</Label>
   234→                <Input
   235→                  id="payment-reference"
   236→                  value={reference}
   237→                  onChange={(e) => setReference(e.target.value)}
   238→                  placeholder="bKash TrxID / Bank ref"
   239→                />
   240→              </div>
   241→            </div>
   242→            <div className="space-y-2">
   243→              <Label htmlFor="payment-notes">Notes</Label>
   244→              <Textarea
   245→                id="payment-notes"
   246→                rows={2}
   247→                value={notes}
   248→                onChange={(e) => setNotes(e.target.value)}
   249→                placeholder="Optional note about this payment"
   250→              />
   251→            </div>
   252→          </div>
   253→        ) : null}
   254→
   255→        <DialogFooter>
   256→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   257→            Cancel
   258→          </Button>
   259→          <Button onClick={() => void submit()} disabled={pending}>
   260→            {pending ? "Saving…" : "Record Payment"}
   261→          </Button>
   262→        </DialogFooter>
   263→      </DialogContent>
   264→    </Dialog>
   265→  )
   266→}

   392→function NewInvoiceDialog({
   393→  open,
   394→  onOpenChange,
   395→  onSaved,
   396→  requireCase = false,
   397→}: {
   398→  open: boolean
   399→  onOpenChange: (open: boolean) => void
   400→  onSaved: () => void
   401→  /** Lawyers must attach the invoice to one of their own cases (server-enforced too). */
   402→  requireCase?: boolean
   403→}) {
   404→  const clientsQ = useApiData<ClientDTO[]>(open ? "/api/clients" : null)
   405→  const casesQ = useApiData<CaseListDTO[]>(open ? "/api/cases" : null)
   406→
   407→  const [clientId, setClientId] = useState("")
   408→  const [caseId, setCaseId] = useState(NO_CASE)
   409→  const [billingType, setBillingType] = useState<string>(BILLING_TYPES[0])
   410→  const [description, setDescription] = useState("")
   411→  const [amount, setAmount] = useState("")
   412→  const [dueDate, setDueDate] = useState("")
   413→  const [pending, setPending] = useState(false)
   414→
   415→  useEffect(() => {
   416→    if (open) {
   417→      setClientId("")
   418→      setCaseId(NO_CASE)
   419→      setBillingType(BILLING_TYPES[0])
   420→      setDescription("")
   421→      setAmount("")
   422→      setDueDate("")
   423→    }
   424→  }, [open])
   425→
   426→  const onCaseChange = (value: string) => {
   427→    setCaseId(value)
   428→    if (value !== NO_CASE && casesQ.data) {
   429→      const kase = casesQ.data.find((c) => c.id === value)
   430→      if (kase?.client?.id) setClientId(kase.client.id)
   431→    }
   432→  }
   433→
   434→  const submit = async () => {
   435→    if (!clientId) {
   436→      toast.error("Please select a client.")
   437→      return
   438→    }
   439→    if (requireCase && (!caseId || caseId === NO_CASE)) {
   440→      toast.error("Please select a case for this invoice.")
   441→      return
   442→    }
   443→    const amt = Number(amount)
   444→    if (!Number.isFinite(amt) || amt <= 0) {
   445→      toast.error("Amount must be greater than 0.")
   446→      return
   447→    }
   448→    setPending(true)
   449→    try {
   450→      await apiSend("POST", "/api/invoices", {
   451→        clientId,
   452→        caseId: caseId !== NO_CASE && caseId ? caseId : undefined,
   453→        billingType,
   454→        description: description.trim() || undefined,
   455→        amount: amt,
   456→        dueDate: dueDate || undefined,
   457→      })
   458→      toast.success("Invoice created")
   459→      onSaved()
   460→      onOpenChange(false)
   461→    } catch (e) {
   462→      toast.error(errorMessage(e))
   463→    } finally {
   464→      setPending(false)
   465→    }
   466→  }
   467→
   468→  return (
   469→    <Dialog open={open} onOpenChange={onOpenChange}>
   470→      <DialogContent className="sm:max-w-lg">
   471→        <DialogHeader>
   472→          <DialogTitle>New Invoice</DialogTitle>
   473→          <DialogDescription>Issue a new invoice to a client.</DialogDescription>
   474→        </DialogHeader>
   475→
   476→        <div className="space-y-4">
   477→          <div className="space-y-2">
   478→            <Label>Client *</Label>
   479→            <Select value={clientId || undefined} onValueChange={setClientId}>
   480→              <SelectTrigger className="w-full">
   481→                <SelectValue placeholder={clientsQ.loading ? "Loading clients…" : "Select client"} />
   482→              </SelectTrigger>
   483→              <SelectContent>
   484→                {(clientsQ.data ?? []).map((c) => (
   485→                  <SelectItem key={c.id} value={c.id}>
   486→                    {c.name}
   487→                  </SelectItem>
   488→                ))}
   489→              </SelectContent>
   490→            </Select>
   491→          </div>
   492→
   493→          <div className="space-y-2">
   494→            <Label>Case (optional)</Label>
   495→            <Select value={caseId} onValueChange={onCaseChange}>
   496→              <SelectTrigger className="w-full">
   497→                <SelectValue placeholder={casesQ.loading ? "Loading cases…" : "No case"} />
   498→              </SelectTrigger>
   499→              <SelectContent>
   500→                <SelectItem value={NO_CASE}>No case (unlinked)</SelectItem>
   501→                {(casesQ.data ?? []).map((c) => (
   502→                  <SelectItem key={c.id} value={c.id}>
   503→                    {c.caseNumber} — {c.title}
   504→                  </SelectItem>
   505→                ))}
   506→              </SelectContent>
   507→            </Select>
   508→            <p className="text-xs text-muted-foreground">Choosing a case auto-selects its client.</p>
   509→          </div>
   510→
   511→          <div className="grid gap-4 sm:grid-cols-2">
   512→            <div className="space-y-2">
   513→              <Label>Billing Type</Label>
   514→              <Select value={billingType} onValueChange={setBillingType}>
   515→                <SelectTrigger className="w-full">
   516→                  <SelectValue />
   517→                </SelectTrigger>
   518→                <SelectContent>
   519→                  {BILLING_TYPES.map((t) => (
   520→                    <SelectItem key={t} value={t}>
   521→                      {t}
   522→                    </SelectItem>
   523→                  ))}
   524→                </SelectContent>
   525→              </Select>
   526→            </div>
   527→            <div className="space-y-2">
   528→              <Label htmlFor="invoice-amount">Amount (৳) *</Label>
   529→              <Input
   530→                id="invoice-amount"
   531→                type="number"
   532→                min="0"
   533→                step="0.01"
   534→                inputMode="decimal"
   535→                value={amount}
   536→                onChange={(e) => setAmount(e.target.value)}
   537→                placeholder="0"
   538→              />
   539→            </div>
   540→          </div>
   541→
   542→          <div className="space-y-2">
   543→            <Label htmlFor="invoice-due">Due Date</Label>
   544→            <Input id="invoice-due" type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} />
   545→          </div>
   546→
   547→          <div className="space-y-2">
   548→            <Label htmlFor="invoice-description">Description</Label>
   549→            <Textarea
   550→              id="invoice-description"
   551→              rows={2}
   552→              value={description}
   553→              onChange={(e) => setDescription(e.target.value)}
   554→              placeholder="What this invoice covers"
   555→            />
   556→          </div>
   557→        </div>
   558→
   559→        <DialogFooter>
   560→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   561→            Cancel
   562→          </Button>
   563→          <Button onClick={() => void submit()} disabled={pending}>
   564→            {pending ? "Creating…" : "Create Invoice"}
   565→          </Button>
   566→        </DialogFooter>
   567→      </DialogContent>
   568→    </Dialog>
   569→  )
   570→}
