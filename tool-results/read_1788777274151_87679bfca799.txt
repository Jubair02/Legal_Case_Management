    59→/** Dhaka-timezone "10:30 AM" for hearing timestamps. */
    60→function formatTime(iso: string | null | undefined): string {
    61→  if (!iso) return "—"
    62→  const d = new Date(iso)
    63→  if (Number.isNaN(d.getTime())) return "—"
    64→  return new Intl.DateTimeFormat("en-GB", {
    65→    timeZone: "Asia/Dhaka",
    66→    hour: "numeric",
    67→    minute: "2-digit",
    68→    hour12: true,
    69→  }).format(d)
    70→}
    71→
    72→/** "THURSDAY" weekday label in Asia/Dhaka. */
    73→function weekdayUpper(iso: string): string {
    74→  const d = new Date(iso)
    75→  if (Number.isNaN(d.getTime())) return ""
    76→  return new Intl.DateTimeFormat("en-GB", { timeZone: "Asia/Dhaka", weekday: "long" })
    77→    .format(d)
    78→    .toUpperCase()
    79→}
    80→
    81→/** Stable Dhaka-day key for grouping ("2026-02-12"). */
    82→function dhakaDayKey(iso: string): string {
    83→  const d = new Date(iso)
    84→  if (Number.isNaN(d.getTime())) return iso
    85→  return new Intl.DateTimeFormat("en-CA", {
    86→    timeZone: "Asia/Dhaka",
    87→    year: "numeric",
    88→    month: "2-digit",
    89→    day: "2-digit",
    90→  }).format(d)
    91→}
    92→
    93→function groupByDay(list: HearingDTO[]): { key: string; date: string; hearings: HearingDTO[] }[] {
    94→  const map = new Map<string, HearingDTO[]>()
    95→  for (const h of list) {
    96→    const key = dhakaDayKey(h.hearingDate)
    97→    const arr = map.get(key)
    98→    if (arr) arr.push(h)
    99→    else map.set(key, [h])
   100→  }
   101→  return Array.from(map.entries()).map(([key, hearings]) => ({ key, date: hearings[0].hearingDate, hearings }))
   102→}

   179→/** Creates a hearing via POST /api/cases/[caseId]/hearings (case picked from active cases). */
   180→export function ScheduleHearingDialog({ open, onOpenChange, onSaved }: ScheduleHearingDialogProps) {
   181→  const [cases, setCases] = useState<CaseListDTO[] | null>(null)
   182→  const [casesLoading, setCasesLoading] = useState(false)
   183→  const [caseQuery, setCaseQuery] = useState("")
   184→  const [selectedCase, setSelectedCase] = useState<CaseListDTO | null>(null)
   185→  const [date, setDate] = useState("")
   186→  const [type, setType] = useState("__none__")
   187→  const [judge, setJudge] = useState("")
   188→  const [court, setCourt] = useState("")
   189→  const [notes, setNotes] = useState("")
   190→  const [pending, setPending] = useState(false)
   191→
   192→  useEffect(() => {
   193→    if (!open) return
   194→    // reset form on every open
   195→    setSelectedCase(null)
   196→    setCaseQuery("")
   197→    setDate("")
   198→    setType("__none__")
   199→    setJudge("")
   200→    setCourt("")
   201→    setNotes("")
   202→    setPending(false)
   203→
   204→    let active = true
   205→    setCasesLoading(true)
   206→    apiGet<CaseListDTO[]>("/api/cases?view=active")
   207→      .then((list) => {
   208→        if (active) setCases(list ?? [])
   209→      })
   210→      .catch((e) => {
   211→        if (active) {
   212→          setCases([])
   213→          toast.error(e instanceof Error ? e.message : "Could not load the case list.")
   214→        }
   215→      })
   216→      .finally(() => {
   217→        if (active) setCasesLoading(false)
   218→      })
   219→    return () => {
   220→      active = false
   221→    }
   222→  }, [open])
   223→
   224→  const candidateCases = useMemo(() => {
   225→    const list = cases ?? []
   226→    const q = caseQuery.trim().toLowerCase()
   227→    if (!q) return list
   228→    return list.filter((c) =>
   229→      [c.caseNumber, c.title, c.client?.name ?? ""].some((v) => v.toLowerCase().includes(q))
   230→    )
   231→  }, [cases, caseQuery])
   232→
   233→  const submit = async () => {
   234→    if (!selectedCase) {
   235→      toast.error("Select the case this hearing belongs to.")
   236→      return
   237→    }
   238→    if (!date) {
   239→      toast.error("Hearing date is required.")
   240→      return
   241→    }

   243→      setPending(true)
   244→      await apiSend<HearingDTO>("POST", `/api/cases/${selectedCase.id}/hearings`, {
   245→        hearingDate: date,
   246→        hearingType: type === "__none__" ? null : type,
   247→        court: court.trim() || null,
   248→        judge: judge.trim() || null,
   249→        notes: notes.trim() || null,
   250→      })
   251→      toast.success(`Hearing scheduled for ${selectedCase.caseNumber}.`)
   252→      onOpenChange(false)
   253→      onSaved()
   254→    } catch (e) {
   255→      toast.error(e instanceof Error ? e.message : "Could not schedule the hearing.")
   256→    } finally {
   257→      setPending(false)
   258→    }
   259→  }
   260→
   261→  return (
   262→    <Dialog open={open} onOpenChange={onOpenChange}>
   263→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   264→        <DialogHeader>
   265→          <DialogTitle>Schedule hearing</DialogTitle>
   266→          <DialogDescription>Pick an active case and set the court date.</DialogDescription>

   268→
   269→        <div className="space-y-4">
   270→          <div className="space-y-1.5">
   271→            <Label>
   272→              Case <span className="text-rose-500">*</span>
   273→            </Label>
   274→            {selectedCase ? (
   275→              <div className="flex items-center justify-between gap-2 rounded-lg border border-emerald-200 bg-emerald-50 px-3 py-2">
   276→                <div className="min-w-0">
   277→                  <p className="truncate text-sm font-semibold text-emerald-900">{selectedCase.caseNumber}</p>
   278→                  <p className="truncate text-xs text-emerald-800/80">{selectedCase.title}</p>
   279→                </div>
   280→                <Button variant="ghost" size="sm" onClick={() => setSelectedCase(null)} disabled={pending}>
   281→                  Change
   282→                </Button>
   283→              </div>
   284→            ) : (
   285→              <div className="space-y-2">

   300→                      {cases && cases.length === 0 ? "No active cases found." : "No cases match your search."}

   392→/** Records the outcome of a hearing via PATCH /api/hearings/[id]. */

   548→  // Lawyers can schedule for their own assigned cases (same as inside a case file).
   549→  const canSchedule = user.role === "ADMIN" || user.role === "STAFF" || user.role === "LAWYER"
   550→  const canUpdate = user.role === "ADMIN" || user.role === "STAFF" || user.role === "LAWYER"
   551→
   552→  const [filter, setFilter] = useState<HearingFilter>("today")
   553→  const { data, loading, error, refetch } = useApiData<HearingDTO[]>(`/api/hearings?filter=${filter}`)
   554→
   555→  const [scheduleOpen, setScheduleOpen] = useState(false)
   556→  const [updateTarget, setUpdateTarget] = useState<HearingDTO | null>(null)
   557→  const [updateOpen, setUpdateOpen] = useState(false)
   558→
   559→  const hearings = useMemo(() => data ?? [], [data])
   560→  const groups = useMemo(
   561→    () => (filter === "today" ? [] : groupByDay(hearings)),
   562→    [filter, hearings]
   563→  )
   564→  const todayGroup = useMemo(() => (filter === "today" ? groupByDay(hearings) : []), [filter, hearings])
   565→
   566→  const openUpdate = (h: HearingDTO) => {
   567→    setUpdateTarget(h)
   568→    setUpdateOpen(true)
   569→  }
   570→
   571→  const renderCards = (list: HearingDTO[]) => (
   572→    <div className="space-y-3">
   573→      {list.map((h) => (
   574→        <HearingCard key={h.id} hearing={h} canUpdate={canUpdate} navigate={navigate} onUpdate={openUpdate} />
   575→      ))}
   576→    </div>
   577→  )
   578→
   579→  const renderDayHeader = (date: string) => {
   580→    const rel = formatRelativeDay(date)
   581→    const isToday = rel === "Today"
   582→    return (
   583→      <div
   584→        className={cn(
   585→          "flex flex-wrap items-baseline gap-x-2 rounded-lg border px-3 py-2",
   586→          isToday ? "border-emerald-200 bg-emerald-50" : "border-stone-200 bg-white"
   587→        )}
   588→      >
   589→        <span className={cn("text-sm font-semibold", isToday ? "text-emerald-800" : "text-foreground")}>
   590→          {isToday ? `Today — ${formatDate(date)}` : formatDate(date)}
   591→        </span>
   592→        <span className="text-[11px] font-medium tracking-wider text-muted-foreground">{weekdayUpper(date)}</span>
   593→        <span className="ml-auto text-[11px] text-muted-foreground">
   594→          {rel !== "Today" && rel !== formatDate(date) ? rel : null}
   595→        </span>
   596→      </div>
   597→    )
   598→  }
   599→
   600→  return (
   601→    <div className="space-y-6">
   602→      <PageHeader title="Hearings" description="Court calendar, hearing updates and adjournment tracking.">
   603→        {canSchedule ? (
   604→          <Button onClick={() => setScheduleOpen(true)}>
   605→            <CalendarPlus className="h-4 w-4" /> Schedule Hearing
   606→          </Button>
   607→        ) : null}
   608→      </PageHeader>
   609→
   610→      <div className="flex w-full overflow-x-auto pb-0.5 sm:w-auto">
   611→        <div className="inline-flex rounded-lg border border-stone-200 bg-white p-1">
   612→          {FILTERS.map((f) => {
   613→            const active = filter === f.key
   614→            return (
   615→              <button
   616→                key={f.key}
   617→                type="button"
   618→                onClick={() => setFilter(f.key)}
   619→                className={cn(
   620→                  "rounded-md px-3 py-1.5 text-sm font-medium whitespace-nowrap transition-colors",
   621→                  active ? "bg-emerald-600 text-white shadow-xs" : "text-muted-foreground hover:bg-stone-100 hover:text-foreground"
   622→                )}
   623→              >
   624→                {f.label}
   625→                {active && data ? ` (${data.length})` : ""}
   626→              </button>
   627→            )
   628→          })}
   629→        </div>
   630→      </div>
   631→
   632→      {loading && !data ? (
   633→        <LoadingBlock rows={4} />
   634→      ) : error && !data ? (
   635→        <EmptyState
   636→          icon={CalendarDays}
   637→          title="Could not load hearings"
   638→          description={error}
   639→          action={
   640→            <Button variant="outline" size="sm" onClick={refetch}>
   641→              Try again
   642→            </Button>
   643→          }
   644→        />
   645→      ) : hearings.length === 0 ? (
   646→        <EmptyState
   647→          icon={CalendarDays}
   648→          title={EMPTY_BY_FILTER[filter].title}
   649→          description={EMPTY_BY_FILTER[filter].description}
   650→          action={
   651→            canSchedule ? (
   652→              <Button size="sm" onClick={() => setScheduleOpen(true)}>
   653→                <CalendarPlus className="h-4 w-4" /> Schedule Hearing
   654→              </Button>
   655→            ) : undefined
   656→          }
   657→        />
   658→      ) : filter === "today" ? (
   659→        <div className="space-y-3">
   660→          {todayGroup.map((g) => (
   661→            <div key={g.key} className="space-y-3">
   662→              {renderDayHeader(g.date)}
   663→              {renderCards(g.hearings)}
   664→            </div>
   665→          ))}
   666→        </div>
   667→      ) : (
   668→        <div className="space-y-6">
   669→          {groups.map((g) => (
   670→            <div key={g.key} className="space-y-3">
   671→              {renderDayHeader(g.date)}
   672→              {renderCards(g.hearings)}
   673→            </div>
   674→          ))}
   675→        </div>
   676→      )}
   677→
   678→      <ScheduleHearingDialog open={scheduleOpen} onOpenChange={setScheduleOpen} onSaved={refetch} />
   679→
   680→      <HearingUpdateDialog
   681→        open={updateOpen}
   682→        onOpenChange={setUpdateOpen}
   683→        hearing={updateTarget}
   684→        onSaved={refetch}
   685→      />
   686→    </div>
   687→  )
   688→}
