    77→export interface CaseFormDialogProps {
    78→  open: boolean

    85→
    86→export function CaseFormDialog({ open, onOpenChange, onSaved, editing, actorRole }: CaseFormDialogProps) {
    87→  const [clients, setClients] = useState<ClientDTO[]>([])
    88→  const [lawyers, setLawyers] = useState<LawyerDTO[]>([])
    89→  const [optionsLoading, setOptionsLoading] = useState(false)
    90→  const [pending, setPending] = useState(false)
    91→
    92→  const [caseNumber, setCaseNumber] = useState("")
    93→  const [title, setTitle] = useState("")
    94→  const [type, setType] = useState<string>("")
    95→  const [clientId, setClientId] = useState<string>("")
    96→  const [lawyerId, setLawyerId] = useState<string>("")
    97→  const [court, setCourt] = useState<string>("")
    98→  const [district, setDistrict] = useState<string>("")
    99→  const [filingDate, setFilingDate] = useState("")
   100→  const [status, setStatus] = useState<string>("ACTIVE")
   101→  const [priority, setPriority] = useState<string>("MEDIUM")
   102→  const [oppositeParty, setOppositeParty] = useState("")
   103→  const [description, setDescription] = useState("")
   104→  const [resolutionSummary, setResolutionSummary] = useState("")
   105→  const [outcome, setOutcome] = useState("")
   106→
   107→  const isEditing = Boolean(editing)
   108→  const isLawyer = actorRole === "LAWYER"
   109→  const closing = status === "RESOLVED" || status === "CLOSED"
   110→  /** Status options: restricted on create, full list when editing. */
   111→  const statusOptions: readonly string[] = isEditing ? CASE_STATUSES : (["DRAFT", "ACTIVE", "PENDING"] as const)
   112→
   113→  // Prefill when the dialog opens
   114→  useEffect(() => {
   115→    if (!open) return
   116→    const e = editing ?? null
   117→    setCaseNumber(e?.caseNumber ?? "")
   118→    setTitle(e?.title ?? "")
   119→    setType(e?.type ?? "")
   120→    setClientId(e?.client?.id ?? "")
   121→    setLawyerId(e?.lawyer?.id ?? "")
   122→    setCourt(e?.court ?? "")
   123→    setDistrict(e?.district ?? "")
   124→    setFilingDate(toDateInputValue(e?.filingDate))
   125→    setStatus(e?.status ?? "ACTIVE")
   126→    setPriority(e?.priority ?? "MEDIUM")
   127→    setOppositeParty(e?.oppositeParty ?? "")
   128→    setDescription(e && "description" in e ? (e.description ?? "") : "")
   129→    setResolutionSummary(e && "resolutionSummary" in e ? (e.resolutionSummary ?? "") : "")
   130→    setOutcome(e && "outcome" in e ? (e.outcome ?? "") : "")
   131→  }, [open, editing])
   132→
   133→  // Load client / lawyer options when opened
   134→  useEffect(() => {
   135→    if (!open) return
   136→    let active = true
   137→    setOptionsLoading(true)
   138→    Promise.all([apiGet<ClientDTO[]>("/api/clients"), apiGet<LawyerDTO[]>("/api/lawyers")])
   139→      .then(([cls, lws]) => {
   140→        if (!active) return
   141→        setClients(Array.isArray(cls) ? cls : [])
   142→        setLawyers(Array.isArray(lws) ? lws : [])
   143→      })
   144→      .catch((e: unknown) => {
   145→        toast.error(e instanceof Error ? e.message : "Could not load clients and lawyers.")
   146→      })
   147→      .finally(() => {
   148→        if (active) setOptionsLoading(false)
   149→      })
   150→    return () => {
   151→      active = false
   152→    }
   153→  }, [open])
   154→
   155→  const handleSubmit = async () => {
   156→    // Client-side validation
   157→    if (!isEditing && !caseNumber.trim()) {
   158→      toast.error("Case number is required.")
   159→      return
   160→    }
   161→    if (!title.trim()) {
   162→      toast.error("Case title is required.")
   163→      return
   164→    }
   165→    if (!type) {
   166→      toast.error("Please select a case type.")
   167→      return
   168→    }
   169→    if (!clientId) {
   170→      toast.error("Please select a client.")
   171→      return
   172→    }
   173→    if (!court) {
   174→      toast.error("Please select a court.")
   175→      return
   176→    }
   177→    if (closing && !resolutionSummary.trim()) {
   178→      toast.error("Resolution summary is required to resolve or close a case.")
   179→      return
   180→    }
   181→
   182→    const prev = editing ?? null
   183→    const payload: Record<string, unknown> = {}
   184→
   185→    if (!prev) {
   186→      // Create — required fields + provided optionals
   187→      payload.caseNumber = caseNumber.trim()
   188→      payload.title = title.trim()
   189→      payload.type = type
   190→      payload.clientId = clientId
   191→      payload.court = court
   192→      if (lawyerId) payload.lawyerId = lawyerId
   193→      if (district) payload.district = district
   194→      if (filingDate) payload.filingDate = filingDate
   195→      payload.status = status
   196→      payload.priority = priority
   197→      if (oppositeParty.trim()) payload.oppositeParty = oppositeParty.trim()
   198→      if (description.trim()) payload.description = description.trim()
   199→      if (closing) payload.resolutionSummary = resolutionSummary.trim()
   200→      if (outcome.trim()) payload.outcome = outcome.trim()
   201→    } else {
   202→      // Edit — only fields that were provided / actually changed.
   203→      // District / filing date / opposite party / description are editable by
   204→      // every role that can open this dialog (the server only blocks
   205→      // assignment changes for lawyers), so they are always diffed and sent.
   206→      payload.title = title.trim()
   207→      payload.type = type
   208→      payload.court = court
   209→      payload.status = status
   210→      payload.priority = priority
   211→      if (!isLawyer) {
   212→        // Only ADMIN/STAFF can change case assignment.
   213→        if (clientId !== prev.client?.id) payload.clientId = clientId
   214→        if ((lawyerId || null) !== (prev.lawyer?.id ?? null)) payload.lawyerId = lawyerId || null
   215→      }
   216→      if (district !== (prev.district ?? "")) payload.district = district.trim() || null
   217→      if ((filingDate || null) !== (prev.filingDate ? toDateInputValue(prev.filingDate) : null))
   218→        payload.filingDate = filingDate || null
   219→      if (oppositeParty.trim() !== (prev.oppositeParty ?? ""))
   220→        payload.oppositeParty = oppositeParty.trim() || null
   221→      const prevDescription = "description" in prev ? (prev.description ?? "") : ""
   222→      if (description.trim() !== prevDescription) payload.description = description.trim() || null
   223→      if (closing) {
   224→        payload.resolutionSummary = resolutionSummary.trim()
   225→        if (outcome.trim()) payload.outcome = outcome.trim()
   226→      }
   227→    }
   228→
   229→    try {
   230→      setPending(true)
   231→      if (prev) {
   232→        await apiSend("PATCH", `/api/cases/${prev.id}`, payload)
   233→        toast.success("Case updated")
   234→      } else {
   235→        await apiSend("POST", "/api/cases", payload)
   236→        toast.success("Case registered")
   237→      }
   238→      onSaved()
   239→      onOpenChange(false)
   240→    } catch (e) {
   241→      toast.error(e instanceof Error ? e.message : "Could not save the case.")
   242→    } finally {
   243→      setPending(false)
   244→    }
   245→  }
   246→
   247→  return (
   248→    <Dialog open={open} onOpenChange={onOpenChange}>
   249→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   250→        <DialogHeader>
   251→          <DialogTitle>{isEditing ? "Edit Case" : "Register New Case"}</DialogTitle>
   252→          <DialogDescription>
   253→            {isEditing
   254→              ? "Update the case file details below."
   255→              : "Fill in the details to register a new case file."}
   256→          </DialogDescription>
   257→        </DialogHeader>
   258→
   259→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   260→          <div className="space-y-2">
   261→            <Label htmlFor="case-number">Case Number</Label>
   262→            <Input
   263→              id="case-number"
   264→              value={caseNumber}
   265→              onChange={(e) => setCaseNumber(e.target.value)}
   266→              placeholder="e.g. CS-124/2026"
   267→              disabled={isEditing}
   268→              className="font-mono"
   269→            />
   270→          </div>
   271→          <div className="space-y-2">
   272→            <Label htmlFor="case-title">
   273→              Case Title <span className="text-rose-600">*</span>
   274→            </Label>
   275→            <Input
   276→              id="case-title"
   277→              value={title}
   278→              onChange={(e) => setTitle(e.target.value)}
   279→              placeholder="e.g. Rahman vs Karim — land dispute"
   280→            />
   281→          </div>
   282→
   283→          <div className="space-y-2">
   284→            <Label>
   285→              Case Type <span className="text-rose-600">*</span>
   286→            </Label>
   287→            <Select value={type || undefined} onValueChange={setType}>
   288→              <SelectTrigger className="w-full">
   289→                <SelectValue placeholder="Select type" />
   290→              </SelectTrigger>
   291→              <SelectContent>
   292→                {CASE_TYPES.map((t) => (
   293→                  <SelectItem key={t} value={t}>
   294→                    {t}
   295→                  </SelectItem>
   296→                ))}
   297→              </SelectContent>
   298→            </Select>
   299→          </div>
   300→
   301→          <div className="space-y-2">
   302→            <Label>
   303→              Client <span className="text-rose-600">*</span>
   304→            </Label>
   305→            <Select
   306→              value={clientId || undefined}
   307→              onValueChange={setClientId}
   308→              disabled={isLawyer}
   309→            >
   310→              <SelectTrigger className="w-full">
   311→                <SelectValue placeholder={optionsLoading ? "Loading clients…" : "Select client"} />
   312→              </SelectTrigger>
   313→              <SelectContent>
   314→                {clients.map((c) => (
   315→                  <SelectItem key={c.id} value={c.id}>
   316→                    {c.name}
   317→                  </SelectItem>
   318→                ))}
   319→              </SelectContent>
   320→            </Select>
   321→          </div>
   322→
   323→          <div className="space-y-2">
   324→            <Label>Assigned Lawyer</Label>
   325→            <Select
   326→              value={lawyerId || NONE}
   327→              onValueChange={(v) => setLawyerId(v === NONE ? "" : v)}
   328→              disabled={isLawyer}
   329→            >
   330→              <SelectTrigger className="w-full">
   331→                <SelectValue placeholder={optionsLoading ? "Loading lawyers…" : "Unassigned"} />
   332→              </SelectTrigger>
   333→              <SelectContent>
   334→                <SelectItem value={NONE}>Unassigned</SelectItem>
   335→                {lawyers.map((l) => (
   336→                  <SelectItem key={l.id} value={l.id}>
   337→                    {l.name}
   338→                  </SelectItem>
   339→                ))}
   340→              </SelectContent>
   341→            </Select>
   342→          </div>
   343→
   344→          <div className="space-y-2">
   345→            <Label>
   346→              Court <span className="text-rose-600">*</span>
   347→            </Label>
   348→            <Select value={court || undefined} onValueChange={setCourt}>
   349→              <SelectTrigger className="w-full">
   350→                <SelectValue placeholder="Select court" />
   351→              </SelectTrigger>
   352→              <SelectContent>
   353→                {COURTS.map((ct) => (
   354→                  <SelectItem key={ct} value={ct}>
   355→                    {ct}
   356→                  </SelectItem>
   357→                ))}
   358→              </SelectContent>
   359→            </Select>
   360→          </div>
   361→
   362→          <div className="space-y-2">
   363→            <Label>District</Label>
   364→            <Select value={district || NONE} onValueChange={(v) => setDistrict(v === NONE ? "" : v)}>
   365→              <SelectTrigger className="w-full">
   366→                <SelectValue placeholder="Select district" />
   367→              </SelectTrigger>
   368→              <SelectContent>
   369→                <SelectItem value={NONE}>Not specified</SelectItem>
   370→                {DISTRICTS.map((d) => (
   371→                  <SelectItem key={d} value={d}>
   372→                    {d}
   373→                  </SelectItem>
   374→                ))}
   375→              </SelectContent>
   376→            </Select>
   377→          </div>
   378→
   379→          <div className="space-y-2">
   380→            <Label htmlFor="filing-date">Filing Date</Label>
   381→            <Input
   382→              id="filing-date"
   383→              type="date"
   384→              value={filingDate}
   385→              onChange={(e) => setFilingDate(e.target.value)}
   386→            />
   387→          </div>
   388→
   389→          <div className="space-y-2">
   390→            <Label>Status</Label>
   391→            <Select value={status} onValueChange={setStatus}>
   392→              <SelectTrigger className="w-full">
   393→                <SelectValue />
   394→              </SelectTrigger>
   395→              <SelectContent>
   396→                {statusOptions.map((s) => (
   397→                  <SelectItem key={s} value={s}>
   398→                    {CASE_STATUS_LABELS[s] ?? s}
   399→                  </SelectItem>
   400→                ))}
   401→              </SelectContent>
   402→            </Select>
   403→          </div>
   404→
   405→          <div className="space-y-2">
   406→            <Label>Priority</Label>
   407→            <Select value={priority} onValueChange={setPriority}>
   408→              <SelectTrigger className="w-full">
   409→                <SelectValue />
   410→              </SelectTrigger>
   411→              <SelectContent>
   412→                {CASE_PRIORITIES.map((p) => (
   413→                  <SelectItem key={p} value={p}>
   414→                    {PRIORITY_LABELS[p] ?? p}
   415→                  </SelectItem>
   416→                ))}
   417→              </SelectContent>
   418→            </Select>
   419→          </div>
   420→
   421→          <div className="space-y-2">
   422→            <Label htmlFor="opposite-party">Opposite Party</Label>
   423→            <Input
   424→              id="opposite-party"
   425→              value={oppositeParty}
   426→              onChange={(e) => setOppositeParty(e.target.value)}
   427→              placeholder="Name of the opposing party"
   428→            />
   429→          </div>
   430→
   431→          <div className="space-y-2 sm:col-span-2">
   432→            <Label htmlFor="case-description">Description</Label>
   433→            <Textarea
   434→              id="case-description"
   435→              value={description}
   436→              onChange={(e) => setDescription(e.target.value)}
   437→              placeholder="Brief summary of the matter…"
   438→              rows={3}
   439→            />
   440→          </div>
   441→
   442→          {closing ? (
   443→            <>
   444→              <div className="space-y-2 sm:col-span-2">
   445→                <Label htmlFor="resolution-summary">
   446→                  Resolution Summary <span className="text-rose-600">*</span>
   447→                </Label>
   448→                <Textarea
   449→                  id="resolution-summary"
   450→                  value={resolutionSummary}
   451→                  onChange={(e) => setResolutionSummary(e.target.value)}
   452→                  placeholder="How was the case resolved?"
   453→                  rows={3}
   454→                  required
   455→                />
   456→                <p className="text-xs text-muted-foreground">
   457→                  Example: Case resolved through mutual settlement.
   458→                </p>
   459→              </div>
   460→              <div className="space-y-2">
   461→                <Label htmlFor="case-outcome">Outcome</Label>
   462→                <Input
   463→                  id="case-outcome"
   464→                  value={outcome}
   465→                  onChange={(e) => setOutcome(e.target.value)}
   466→                  placeholder="e.g. Won, Settled, Dismissed"
   467→                />
   468→              </div>
   469→            </>
   470→          ) : null}
   471→        </div>
   472→
   473→        <DialogFooter>
   474→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   475→            Cancel
   476→          </Button>
   477→          <Button onClick={() => void handleSubmit()} disabled={pending}>
   478→            {pending ? <Loader2 className="h-4 w-4 animate-spin" /> : null}
   479→            {isEditing ? "Save Changes" : "Register Case"}
   480→          </Button>
   481→        </DialogFooter>
   482→      </DialogContent>
   483→    </Dialog>
   484→  )
   485→}

   491→export default function CasesView({ user, navigate }: ViewProps) {
   492→  const [view, setView] = useState<ViewTab>("all")
   493→  const [status, setStatus] = useState("")
   494→  const [type, setType] = useState("")
   495→  const [priority, setPriority] = useState("")
   496→  const [searchInput, setSearchInput] = useState("")
   497→  const [search, setSearch] = useState("")
   498→  const [formOpen, setFormOpen] = useState(false)
   499→
   500→  // 300ms debounce for the search box
   501→  useEffect(() => {
   502→    const t = setTimeout(() => setSearch(searchInput.trim()), 300)
   503→    return () => clearTimeout(t)
   504→  }, [searchInput])
   505→
   506→  const url = useMemo(() => {
   507→    const qs = new URLSearchParams()
   508→    qs.set("view", view)
   509→    if (status) qs.set("status", status)
   510→    if (type) qs.set("type", type)
   511→    if (priority) qs.set("priority", priority)
   512→    if (search) qs.set("search", search)
   513→    return `/api/cases?${qs.toString()}`
   514→  }, [view, status, type, priority, search])
   515→
   516→  const { data, loading, error, refetch } = useApiData<CaseListDTO[]>(url)
   517→  const rows = useMemo(() => (Array.isArray(data) ? data : []), [data])
   518→
   519→  const canCreate = user.role === "ADMIN" || user.role === "STAFF"
   520→  const hasFilters = Boolean(status || type || priority || searchInput)
   521→
   522→  const clearFilters = () => {
   523→    setStatus("")
   524→    setType("")
   525→    setPriority("")
   526→    setSearchInput("")
   527→  }
   528→
   529→  const loadingFirst = loading && !data && !error
   530→
   531→  return (
   532→    <div className="space-y-6">
   533→      <PageHeader
   534→        title="Case Management"
   535→        description="Register, track and resolve cases across Bangladesh courts."
   536→      >
   537→        <div className="flex items-center rounded-lg border border-stone-200/80 bg-white p-0.5">
   538→          {VIEW_TABS.map((tab) => (
   539→            <Button
   540→              key={tab.key}
   541→              size="sm"
   542→              variant={view === tab.key ? "default" : "outline"}
   543→              className={cn("h-7 border-0 px-3", view !== tab.key && "bg-transparent shadow-none")}
   544→              onClick={() => setView(tab.key)}
   545→            >
   546→              {tab.label}
   547→            </Button>
   548→          ))}
   549→        </div>
   550→        {canCreate ? (
   551→          <Button onClick={() => setFormOpen(true)}>
   552→            <Plus className="h-4 w-4" /> New Case
   553→          </Button>
   554→        ) : null}
   555→      </PageHeader>
   556→
   557→      {/* Filter row */}
   558→      <div className="flex flex-wrap items-center gap-2">
   559→        <div className="relative min-w-[15rem] flex-1 sm:max-w-xs">
   560→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   561→          <Input
   562→            value={searchInput}
   563→            onChange={(e) => setSearchInput(e.target.value)}
   564→            placeholder="Search case no, title, opposite party…"
   565→            aria-label="Search cases"
   566→            className="pl-8"
   567→          />
   568→        </div>
   569→        <Select value={status || "all"} onValueChange={(v) => setStatus(v === "all" ? "" : v)}>
   570→          <SelectTrigger className="w-[10.5rem]">
   571→            <SelectValue placeholder="All statuses" />
   572→          </SelectTrigger>
   573→          <SelectContent>
   574→            <SelectItem value="all">All statuses</SelectItem>
   575→            {Object.entries(CASE_STATUS_LABELS).map(([key, label]) => (
   576→              <SelectItem key={key} value={key}>
   577→                {label}
   578→              </SelectItem>
   579→            ))}
   580→          </SelectContent>
   581→        </Select>
   582→        <Select value={type || "all"} onValueChange={(v) => setType(v === "all" ? "" : v)}>
   583→          <SelectTrigger className="w-[11.5rem]">
   584→            <SelectValue placeholder="All types" />

   719→          </CardContent>

   722→
   723→      <CaseFormDialog
