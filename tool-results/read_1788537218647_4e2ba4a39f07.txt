    14→    14→/** Minimal valid PDF generator (single page, Helvetica). */
    15→    15→function makePdf(lines: string[]): Buffer {
    16→    16→  const content = lines
    17→    17→    .map((l, i) => {
    18→    18→      const text = l.replace(/[\\()]/g, "").slice(0, 90)
    19→    19→      const y = 740 - i * 22
    20→    20→      const size = i === 0 ? 15 : 11
    21→    21→      return `BT /F1 ${size} Tf 60 ${y} Td (${text}) Tj ET`
    22→    22→    })
    23→    23→    .join("\n")
    24→    24→  const objects = [
    25→    25→    "<< /Type /Catalog /Pages 2 0 R >>",
    26→    26→    "<< /Type /Pages /Kids [3 0 R] /Count 1 >>",
    27→    27→    "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Contents 4 0 R /Resources << /Font << /F1 5 0 R >> >> >>",
    28→    28→    `<< /Length ${content.length} >>\nstream\n${content}\nendstream`,
    29→    29→    "<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>",
    30→    30→  ]
    31→    31→  let pdf = "%PDF-1.4\n"
    32→    32→  const offsets: number[] = []
    33→    33→  objects.forEach((body, i) => {
    34→    34→    offsets.push(pdf.length)
    35→    35→    pdf += `${i + 1} 0 obj\n${body}\nendobj\n`
    36→    36→  })
    37→    37→  const xrefStart = pdf.length
    38→    38→  pdf += `xref\n0 ${objects.length + 1}\n0000000000 65535 f \n`
    39→    39→  offsets.forEach((o) => {
    40→    40→    pdf += `${String(o).padStart(10, "0")} 00000 n \n`
    41→    41→  })
    42→    42→  pdf += `trailer\n<< /Size ${objects.length + 1} /Root 1 0 R >>\nstartxref\n${xrefStart}\n%%EOF`
    43→    43→  return Buffer.from(pdf, "latin1")
    44→    44→}

    46→    46→/** Dhaka civil time helper: dayOffset 0 = today, hour in Dhaka local time. */
    47→    47→function dhakaAt(dayOffset: number, hour = 10): Date {
    48→    48→  const now = new Date()
    49→    49→  const base = new Date(now.getTime() + dayOffset * 24 * 60 * 60 * 1000)
    50→    50→  const y = base.getUTCFullYear()
    51→    51→  const m = String(base.getUTCMonth() + 1).padStart(2, "0")
    52→    52→  const d = String(base.getUTCDate()).padStart(2, "0")
    53→    53→  const utcHour = hour - 6 // Dhaka = UTC+6 (no DST)
    54→    54→  return new Date(`${y}-${m}-${d}T${String(utcHour).padStart(2, "0")}:00:00.000Z`)
    55→    55→}
    56→    56→
    57→    57→function dateOnly(dayOffset: number): Date {
    58→    58→  return dhakaAt(dayOffset, 12)
    59→    59→}
