   264→export interface LawyerFormDialogProps {
   265→  open: boolean
   266→  onOpenChange: (open: boolean) => void
   267→  /** When set the dialog edits this lawyer; otherwise it creates a new one. */
   268→  lawyer?: LawyerDTO | null
   269→  onSaved: () => void
   270→}
   271→
   272→export function LawyerFormDialog({ open, onOpenChange, lawyer, onSaved }: LawyerFormDialogProps) {
   273→  const isEdit = Boolean(lawyer)
   274→  const [name, setName] = useState("")
   275→  const [phone, setPhone] = useState("")
   276→  const [email, setEmail] = useState("")
   277→  const [barCouncilId, setBarCouncilId] = useState("")
   278→  const [specialization, setSpecialization] = useState("__none__")
   279→  const [chamberName, setChamberName] = useState("")
   280→  const [experience, setExperience] = useState("")
   281→  const [status, setStatus] = useState("ACTIVE")
   282→  const [portal, setPortal] = useState(false)
   283→  const [password, setPassword] = useState("")
   284→  const [pending, setPending] = useState(false)
   285→
   286→  useEffect(() => {
   287→    if (!open) return
   288→    setName(lawyer?.name ?? "")
   289→    setPhone(lawyer?.phone ?? "")
   290→    setEmail(lawyer?.email ?? "")
   291→    setBarCouncilId(lawyer?.barCouncilId ?? "")
   292→    setSpecialization(lawyer?.specialization && SPECIALIZATIONS.includes(lawyer.specialization as (typeof SPECIALIZATIONS)[number]) ? lawyer.specialization : "__none__")
   293→    setChamberName(lawyer?.chamberName ?? "")
   294→    setExperience(lawyer?.experience != null ? String(lawyer.experience) : "")
   295→    setStatus(lawyer?.status === "INACTIVE" ? "INACTIVE" : "ACTIVE")
   296→    setPortal(false)
   297→    setPassword("")
   298→    setPending(false)
   299→  }, [open, lawyer])
   300→
   301→  const submit = async () => {
   302→    if (!name.trim()) {
   303→      toast.error("Lawyer name is required.")
   304→      return
   305→    }
   306→    const expNum = experience.trim() === "" ? null : Number(experience)
   307→    if (expNum !== null && (Number.isNaN(expNum) || expNum < 0)) {
   308→      toast.error("Experience must be a non-negative number.")
   309→      return
   310→    }
   311→    if (!isEdit && portal) {
   312→      if (!email.trim()) {
   313→        toast.error("Email is required to create a portal account.")
   314→        return
   315→      }
   316→      if (password.length < 6) {
   317→        toast.error("Portal password must be at least 6 characters.")
   318→        return
   319→      }
   320→    }
   321→    const base = {
   322→      name: name.trim(),
   323→      phone: phone.trim() || null,
   324→      email: email.trim() || null,
   325→      barCouncilId: barCouncilId.trim() || null,
   326→      specialization: specialization === "__none__" ? null : specialization,
   327→      chamberName: chamberName.trim() || null,
   328→      experience: expNum,
   329→    }
   330→    try {
   331→      setPending(true)
   332→      if (isEdit && lawyer) {
   333→        await apiSend<LawyerDTO>("PATCH", `/api/lawyers/${lawyer.id}`, { ...base, status })
   334→        toast.success("Lawyer updated.")
   335→      } else {
   336→        await apiSend<LawyerDTO>("POST", "/api/lawyers", {
   337→          ...base,
   338→          ...(portal ? { createPortalAccess: true, password } : {}),
   339→        })
   340→        toast.success(portal ? "Lawyer created with portal access." : "Lawyer created.")
   341→      }
   342→      onOpenChange(false)
   343→      onSaved()
   344→    } catch (e) {
   345→      toast.error(e instanceof Error ? e.message : "Could not save the lawyer.")
   346→    } finally {
   347→      setPending(false)
   348→    }
   349→  }
   350→
   351→  return (
   352→    <Dialog open={open} onOpenChange={onOpenChange}>
   353→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   354→        <DialogHeader>
   355→          <DialogTitle>{isEdit ? "Edit lawyer" : "Add lawyer"}</DialogTitle>
   356→          <DialogDescription>
   357→            {isEdit ? "Update the advocate's credentials and status." : "Register an advocate of the chamber."}
   358→          </DialogDescription>
   359→        </DialogHeader>
   360→
   361→        <div className="space-y-4">
   362→          <div className="space-y-1.5">
   363→            <Label htmlFor="lawyer-name">
   364→              Name <span className="text-rose-500">*</span>
   365→            </Label>
   366→            <Input
   367→              id="lawyer-name"
   368→              value={name}
   369→              onChange={(e) => setName(e.target.value)}
   370→              placeholder="e.g. Kamal Hossain"
   371→              autoFocus
   372→            />
   373→          </div>
   374→
   375→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   376→            <div className="space-y-1.5">
   377→              <Label htmlFor="lawyer-phone">Phone</Label>
   378→              <Input id="lawyer-phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+8801XXXXXXXXX" />
   379→            </div>
   380→            <div className="space-y-1.5">
   381→              <Label htmlFor="lawyer-email">Email</Label>
   382→              <Input id="lawyer-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="lawyer@chamber.bd" />
   383→            </div>
   384→            <div className="space-y-1.5">
   385→              <Label htmlFor="lawyer-bar">Bar Council ID</Label>
   386→              <Input id="lawyer-bar" value={barCouncilId} onChange={(e) => setBarCouncilId(e.target.value)} placeholder="e.g. D-12345" />
   387→            </div>
   388→            <div className="space-y-1.5">
   389→              <Label htmlFor="lawyer-specialization">Specialization</Label>
   390→              <Select value={specialization} onValueChange={setSpecialization}>
   391→                <SelectTrigger id="lawyer-specialization" className="w-full">
   392→                  <SelectValue placeholder="Not specified" />
   393→                </SelectTrigger>
   394→                <SelectContent>
   395→                  <SelectItem value="__none__">Not specified</SelectItem>
   396→                  {SPECIALIZATIONS.map((s) => (
   397→                    <SelectItem key={s} value={s}>
   398→                      {s}
   399→                    </SelectItem>
   400→                  ))}
   401→                </SelectContent>
   402→              </Select>
   403→            </div>
   404→            <div className="space-y-1.5">
   405→              <Label htmlFor="lawyer-chamber">Chamber name</Label>
   406→              <Input id="lawyer-chamber" value={chamberName} onChange={(e) => setChamberName(e.target.value)} placeholder="e.g. Hossain & Associates" />
   407→            </div>
   408→            <div className="space-y-1.5">
   409→              <Label htmlFor="lawyer-experience">Experience (years)</Label>
   410→              <Input
   411→                id="lawyer-experience"
   412→                type="number"
   413→                min={0}
   414→                value={experience}
   415→                onChange={(e) => setExperience(e.target.value)}
   416→                placeholder="e.g. 12"
   417→              />
   418→            </div>
   419→            {isEdit ? (
   420→              <div className="space-y-1.5">
   421→                <Label htmlFor="lawyer-status">Status</Label>
   422→                <Select value={status} onValueChange={setStatus}>
   423→                  <SelectTrigger id="lawyer-status" className="w-full">
   424→                    <SelectValue placeholder="Select status" />
   425→                  </SelectTrigger>
   426→                  <SelectContent>
   427→                    <SelectItem value="ACTIVE">Active</SelectItem>
   428→                    <SelectItem value="INACTIVE">Inactive</SelectItem>
   429→                  </SelectContent>
   430→                </Select>
   431→              </div>
   432→            ) : null}
   433→          </div>
   434→
   435→          {!isEdit ? (
   436→            <div className="space-y-3 rounded-lg border border-stone-200 bg-stone-50 p-3">
   437→              <div className="flex items-center justify-between gap-3">
   438→                <div className="min-w-0">
   439→                  <Label htmlFor="lawyer-portal">Create lawyer portal account</Label>
   440→                  <p className="text-xs text-muted-foreground">Lawyer signs in with their email + this password</p>
   441→                </div>
   442→                <Switch id="lawyer-portal" checked={portal} onCheckedChange={setPortal} />
   443→              </div>
   444→              {portal ? (
   445→                <div className="space-y-1.5">
   446→                  <Label htmlFor="lawyer-portal-password">Portal password</Label>
   447→                  <Input
   448→                    id="lawyer-portal-password"
   449→                    type="password"
   450→                    value={password}
   451→                    onChange={(e) => setPassword(e.target.value)}
   452→                    placeholder="Minimum 6 characters"
   453→                    autoComplete="new-password"
   454→                  />
   455→                  <p className="text-xs text-muted-foreground">Minimum 6 characters.</p>
   456→                </div>
   457→              ) : null}
   458→            </div>
   459→          ) : null}
   460→        </div>
   461→
   462→        <DialogFooter>
   463→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={pending}>
   464→            Cancel
   465→          </Button>
   466→          <Button onClick={submit} disabled={pending}>
   467→            {pending ? "Saving…" : isEdit ? "Save changes" : "Create lawyer"}
   468→          </Button>
   469→        </DialogFooter>
   470→      </DialogContent>
   471→    </Dialog>
   472→  )
   473→}
